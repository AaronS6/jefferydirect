    22→function totals(items: { qty: number; unitPriceCents: number }[]) {
    23→  const subtotal = items.reduce((s, i) => s + i.qty * i.unitPriceCents, 0);
    24→  const tax = Math.round((subtotal * TAX_BP) / 10_000);
    25→  return { subtotal, tax, total: subtotal + tax };
