   499→function UserRow({
   500→  user,
   501→  customerId,
   502→  onDone,
   503→}: {
   504→  user: CustomerProfile["users"][number];
   505→  customerId: string;
   506→  onDone: () => void;
   507→}) {
   508→  const queryClient = useQueryClient();
   509→  const [confirmDisable, setConfirmDisable] = useState(false);
   510→
   511→  const roleMutation = useMutation({
   512→    mutationFn: (orgRole: string) =>
   513→      api.patch<{ ok: boolean }>(`/api/admin/customers/${customerId}/users/${user.id}`, { orgRole }),
   514→    onSuccess: (_res, orgRole) => {
   515→      queryClient.invalidateQueries({ queryKey: ["admin", "customer", customerId] });
   516→      queryClient.invalidateQueries({ queryKey: ["admin", "customers"] });
   517→      onDone();
   518→      toast({
   519→        title: `${user.name} is now ${ORG_ROLE_LABELS[orgRole] ?? orgRole}`,
   520→      });
   521→    },
   522→    onError: (err: Error) => toast({ title: "Could not change role", description: err.message }),
   523→  });
   524→
   525→  const toggle = useMutation({
   526→    mutationFn: (status: "ACTIVE" | "DISABLED") =>
