    83→function nextStatusAfter(status: string): string | undefined {
    84→  const i = ORDER_STATUSES.indexOf(status as (typeof ORDER_STATUSES)[number]);
    85→  if (i === -1 || i === ORDER_STATUSES.length - 1) return undefined;
    86→  return ORDER_STATUSES[i + 1];
    87→}

    89→/** Date-ish ISO → yyyy-mm-dd for <input type="date"> controls. */
    90→function toDateInput(iso: string | null | undefined): string {
    91→  if (!iso) return "";
    92→  const m = /^\d{4}-\d{2}-\d{2}/.exec(iso);
    93→  return m ? m[0] : "";
    94→}

    96→/** One human-readable sentence for a single requested change. */
    97→function describeChange(item: ChangeItem): string {
    98→  switch (item.kind) {
    99→    case "QUANTITY": {
   100→      const label = item.name ?? item.sku ?? "product";
   101→      return `Quantity — ${label}: ${(item.currentQty ?? 0).toLocaleString()} → ${(item.newQty ?? 0).toLocaleString()}`;
   102→    }
   103→    case "SPEC": {
   104→      const label = item.name ?? item.sku;
   105→      return `Specification — ${label ? `${label}: ` : ""}${item.spec ?? "—"}`;
   106→    }
   107→    case "ADD":
   108→      return `Add ${item.name ?? "product"}${item.newQty ? ` × ${item.newQty.toLocaleString()}` : ""}`;
   109→    case "REMOVE":
   110→      return `Remove ${item.name ?? item.sku ?? "product"}`;
   111→    case "ADDRESS":
   112→      return `Delivery address: ${item.newAddress ?? "—"}`;
   113→    case "DATE":
   114→      return `Delivery date → ${item.newDate ? formatDate(item.newDate) : "—"}`;
   115→    case "CANCEL":
   116→      return "Cancel order";
   117→    case "OTHER":
   118→      return `Other: ${item.note ?? "—"}`;
   119→  }
   120→}

   134→function SpecChips({ specs }: { specs: LineSpecs }) {
   135→  const entries = (Object.entries(specs) as [keyof LineSpecs, string | undefined][])
   136→    .filter(([, v]) => v && v.trim())
   137→    .slice(0, 8);
   138→  if (entries.length === 0) return null;
   139→  return (
   140→    <ul className="mt-1.5 flex flex-wrap gap-1.5">
   141→      {entries.map(([key, value]) => (
   142→        <li
   143→          key={key}
   144→          className="inline-flex max-w-full items-center rounded-full border bg-card px-2 text-caption text-muted-foreground"
   145→          title={value}
   146→        >
   147→          {key === "custom" ? (
   148→            <span className="truncate">{value}</span>
   149→          ) : (
   150→            <span className="truncate">
   151→              {SPEC_LABELS[key] ?? key}: {value}
   152→            </span>
   153→          )}
   154→        </li>
   155→      ))}
   156→    </ul>
   157→  );
   158→}

  1091→  const queryClient = useQueryClient();

  1093→  const [declineOpen, setDeclineOpen] = useState(false);

  1096→  const [counterQty, setCounterQty] = useState<Record<number, string>>(() =>
  1097→    Object.fromEntries(
  1098→      request.changes
  1099→        .map((c, i) => (c.kind === "QUANTITY" && c.newQty != null ? [i, String(c.newQty)] : null))
  1100→        .filter((v): v is [number, string] => v !== null),
  1101→    ),
  1102→  );
  1103→  const [counterDate, setCounterDate] = useState<string>(() => {
  1104→    const dateItem = request.changes.find((c) => c.kind === "DATE" && c.newDate);
  1105→    return dateItem?.newDate ? toDateInput(dateItem.newDate) : "";
  1106→  });
  1107→  const [counterNote, setCounterNote] = useState("");
  1108→  const [counterError, setCounterError] = useState<string | null>(null);
  1109→
  1110→  const decide = useMutation({
  1111→    mutationFn: (payload: { decision: "ACCEPT" | "DECLINE" | "COUNTER"; note?: string; counter?: ChangeItem[] }) =>
  1112→      api.post<{ ok: boolean }>(`/api/admin/change-requests/${request.id}/decide`, payload),
  1113→    onSuccess: (_res, payload) => {
  1114→      queryClient.invalidateQueries({ queryKey: ["admin", "order", request.orderId] });
  1115→      queryClient.invalidateQueries({ queryKey: ["admin", "orders"] });
  1116→      queryClient.invalidateQueries({ queryKey: ["admin", "attention"] });
  1117→      queryClient.invalidateQueries({ queryKey: ["notifications"] });

  1120→      setDeclineOpen(false);
  1121→      setDeclineNote("");

  1124→        title:
  1125→          payload.decision === "ACCEPT"
  1126→            ? "Change approved and applied"
  1127→            : payload.decision === "DECLINE"
  1128→              ? "Change request declined"
  1129→              : "Alternative proposed",
  1130→        description:
  1131→          payload.decision === "ACCEPT"
  1132→            ? "The order now reflects the requested change; the original values stay preserved in this request's history."
  1133→            : payload.decision === "DECLINE"

  1143→  function buildCounter(): ChangeItem[] | null {

  1145→      if (c.kind === "QUANTITY" && counterQty[i] !== undefined) {
  1146→        const qty = parseInt(counterQty[i], 10);
  1147→        if (!Number.isFinite(qty) || qty < 1) return c;
  1148→        return { ...c, newQty: qty };
  1149→      }
  1150→      if (c.kind === "DATE" && counterDate) return { ...c, newDate: counterDate };
  1151→      return c;
  1152→    });
  1153→    const badQty = request.changes.some((c, i) => {
  1154→      if (c.kind !== "QUANTITY" || counterQty[i] === undefined) return false;
  1155→      const qty = parseInt(counterQty[i], 10);
  1156→      return !Number.isFinite(qty) || qty < 1;
  1157→    });
  1158→    if (badQty) {
  1159→      setCounterError("Quantities must be whole numbers of at least 1.");
  1160→      return null;
  1161→    }
  1162→    setCounterError(null);
  1163→    return items;
  1164→  }

  1169→        <ChangeStatusBadge status={request.status} />

  1171→          {request.requestedByName}
  1172→          <span className="ml-1 font-normal text-muted-foreground">requested {timeAgo(request.createdAt)}</span>
  1173→        </p>
  1174→        {request.appliedAt && (
  1175→          <span className="rounded-full bg-[var(--status-done-bg)] px-1.5 py-0.5 text-micro font-medium text-[var(--status-done-fg)]">
  1176→            applied
  1177→          </span>

  1181→      <div className="space-y-3 px-4 py-3.5 sm:px-4">
