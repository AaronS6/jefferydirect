    50→// ─── Builder draft state ─────────────────────────────────────────────────────
    51→
    52→type DraftRow = { uid: string; description: string; kind: string; qty: string; price: string };
    53→
    54→let uidCounter = 0;
    55→const newUid = () => `row-${++uidCounter}`;
    56→
    57→function rowsFromQuote(quote: QuotationDetail): DraftRow[] {
    58→  if (quote.items.length === 0) return [emptyRow()];
    59→  return quote.items.map((i) => ({
    60→    uid: newUid(),
    61→    description: i.description,
    62→    kind: i.kind,
    63→    qty: String(i.qty),
    64→    price: centsToInputValue(i.unitPriceCents),
    65→  }));
    66→}
    67→
    68→function emptyRow(): DraftRow {
    69→  return { uid: newUid(), description: "", kind: "PRODUCT", qty: "", price: "" };
    70→}
    71→
    72→function rowCents(row: DraftRow): number | null {
    73→  const cents = parseDollarsToCents(row.price.trim());
    74→  const qty = Number.parseInt(row.qty, 10);
    75→  if (cents === null || !Number.isFinite(qty) || qty <= 0) return null;
    76→  return cents * qty;
    77→}
    78→
    79→function rowIssues(row: DraftRow): { description: boolean; qty: boolean; price: boolean } {
    80→  const cents = parseDollarsToCents(row.price.trim());
    81→  const qty = Number.parseInt(row.qty, 10);
    82→  return {
    83→    description: row.description.trim() === "",
    84→    qty: !Number.isFinite(qty) || qty <= 0,
    85→    price: cents === null,
    86→  };
    87→}

   136→        title: "Request acknowledged",

   147→        kind: r.kind,

   152→        items,
   153→        expiryDate: expiry || null,
   154→        terms: terms.trim() || null,
   155→        internalNote: internalNote.trim() || null,
   156→      });

   164→        description: "The customer can now review pricing and respond from their portal.",

   166→    },

   173→        <PageHeader title="Quotation" onBack={back} eyebrow="Quotation detail" />

   176→    );
   177→  }
   178→
   179→  if (isLoading || !quote) {
   180→    return (
   181→      <div>

   183→        <div className="grid gap-4 sm:gap-6 lg:grid-cols-[1fr_360px]">
   184→          <SectionCard>
   185→            <RowsSkeleton rows={5} />
   186→          </SectionCard>
   187→          <SectionCard>
   188→            <RowsSkeleton rows={4} />
   189→          </SectionCard>
   190→        </div>
   191→      </div>
   192→    );
   193→  }
   194→

   196→  const reissuable = quote.status === "ISSUED" || quote.status === "REVISED" || quote.status === "EXPIRED";

   226→        title={quote.title}

   232→              className="underline-offset-4 transition-colors hover:text-primary hover:underline focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring rounded"
   233→              onClick={() => navigate("admin-customer-detail", { id: quote.orgId })}

   235→              {quote.orgName}

   237→            <span className="text-muted-foreground/50">·</span>

   239→            <span className="text-muted-foreground/50">·</span>
   240→            <span>{formatDate(quote.createdAt)}</span>
   241→          </span>
   242→        }
   243→        actions={
   244→          <div className="flex flex-wrap items-center gap-2">

   246→              <span className="rounded-md border bg-muted/40 px-2 py-0.5 text-[11.5px] font-medium text-muted-foreground">
   247→                Revision {quote.revision}
   248→              </span>

   250→            {preIssue && (

   252→                size="sm"

   254→                className="h-8 gap-1.5 text-[12.5px]"
   255→                disabled={quote.status !== "REQUESTED" || acknowledge.isPending}
   256→                onClick={() => acknowledge.mutate()}

   258→                {acknowledge.isPending ? <Loader2 className="size-3.5 animate-spin" aria-hidden /> : <CheckCircle2 className="size-3.5" aria-hidden />}
   259→                Acknowledge
   260→              </Button>
   261→            )}
   262→            <StatusBadge status={quote.status} kind="quote" />
   263→          </div>
   264→        }
   265→      />
   266→
   267→      <div className="grid items-start gap-4 sm:gap-6 lg:grid-cols-[minmax(0,1fr)_360px]">
   268→        {/* Zone A + issued quote + builder */}

   285→              <DefRow label="Comments" value={dashed(quote.comments)} />

   287→          </SectionCard>
   288→
   289→          {/* Response zone */}

   291→            <div
   292→              className={cn(

   295→              )}
   296→            >
   297→              {quote.status === "ACCEPTED" ? (
   298→                <CheckCircle2 className="mt-0.5 size-5 shrink-0 text-emerald-600" aria-hidden />
   299→              ) : (
   300→                <XCircle className="mt-0.5 size-5 shrink-0 text-red-600" aria-hidden />

   304→                  Customer {quote.status === "ACCEPTED" ? "accepted" : "declined"} this quotation
   305→                  {quote.respondedAt ? ` · ${formatDateTime(quote.respondedAt)}` : ""}
   306→                </p>
   307→                {quote.respondedNote && (
   308→                  <p className="mt-1 text-[13px] leading-relaxed text-foreground/80">{quote.respondedNote}</p>
   309→                )}
   310→              </div>
   311→            </div>

   315→            <div className="flex items-start gap-3 rounded-lg border border-border bg-muted/40 p-4 text-[13px] leading-relaxed text-muted-foreground">

   318→                This quotation expired{quote.expiryDate ? ` on ${formatDate(quote.expiryDate)}` : ""} without a customer
   319→                response. Re-issue to send fresh pricing.
   320→              </p>

   323→
   324→          {/* Issued quotation view */}
   325→          {hasItems && (reissuable || responded || quote.status === "EXPIRED") && (

   327→              title={quote.status === "REVISED" ? `Issued quotation — revision ${quote.revision}` : "Issued quotation"}
   328→              description={
   329→                quote.expiryDate ? `Issued ${formatDate(quote.issuedAt)} · valid until ${formatDate(quote.expiryDate)}` : `Issued ${formatDate(quote.issuedAt)}`
   330→              }
   331→              action={
   332→                <div className="flex flex-wrap gap-2">
   333→                  <Button
   334→                    variant="outline"
   335→                    size="sm"
   336→                    className="h-8 gap-1.5 text-[12.5px]"
   337→                    onClick={() => setPreviewOpen(true)}
   338→                  >
   339→                    <Eye className="size-3.5" aria-hidden /> Preview document
   340→                  </Button>
   341→                  {reissuable && (
   342→                    <Button
   343→                      variant={preIssue ? "default" : "outline"}
   344→                      size="sm"
   345→                      className="h-8 gap-1.5 text-[12.5px]"
   346→                      onClick={() => {
   347→                        setBuilderOpen(false);
   348→                        openBuilderPrefilled(quote);
   349→                      }}
   350→                    >
   351→                      <RefreshCcw className="size-3.5" aria-hidden /> Re-issue (revision)

   353→                  )}
   354→                </div>
   355→              }
   356→            >
   357→              <Table>
   358→                <TableHeader>
   359→                  <TableRow>
   360→                    <Th>Description</Th>
   361→                    <Th>Type</Th>
