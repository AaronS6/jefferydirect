    79→// Order progress hero — the full nine-step rail at xl and up; a compact
    80→// progress bar below xl (the vertical timeline below carries full history).
    81→// Ink dots for completed steps, kraft ring on the current one, hollow future.
    82→function StatusStepper({ status, history }: { status: string; history: OrderHistoryEntry[] }) {
    83→  const flow = ORDER_STATUSES;
    84→  const terminal = ORDER_TERMINAL_STATUSES.includes(status);
    85→  const currentIdx = flow.indexOf(status as (typeof flow)[number]);
    86→
    87→  // Latest history entry per status (ascending order).
    88→  const latestByStatus = new Map<string, OrderHistoryEntry>();
    89→  for (const h of [...history].sort((a, b) => a.createdAt.localeCompare(b.createdAt))) {
    90→    latestByStatus.set(h.status, h);
    91→  }
    92→
    93→  const progress =
    94→    currentIdx >= 0 ? Math.round(((currentIdx + 1) / flow.length) * 100) : 100;
    95→  const dead = status === "CANCELLED" || status === "REJECTED";
    96→
    97→  return (
    98→    <>
    99→      {/* Compact progress — below xl */}
   100→      <div className="flex flex-wrap items-center gap-x-4 gap-y-2.5 xl:hidden" aria-label="Order progress">
   101→        <span
   102→          aria-hidden
   103→          className={cn(
   104→            "flex size-[14px] shrink-0 items-center justify-center rounded-full",
   105→            dead
   106→              ? "border-2 border-[var(--status-danger-dot)] bg-[var(--status-danger-bg)]"
   107→              : terminal
   108→                ? "bg-foreground"
   109→                : "border-2 border-kraft-strong bg-kraft",
   110→          )}
   111→        >
   112→          {!dead && !terminal && <span className="size-1.5 rounded-full bg-background" aria-hidden />}
   113→          {terminal && <span className="size-[4.5px] rounded-full bg-background" aria-hidden />}
   114→        </span>
   115→        <p className="text-body font-semibold text-foreground">
   116→          {ORDER_STATUS_LABELS[status] ?? status}
   117→        </p>
   118→        {!dead && (
   119→          <p className="text-caption text-muted-foreground">
   120→            <span className="tnum">Step {Math.max(1, currentIdx + 1)} of {flow.length}</span>
   121→          </p>
   122→        )}
   123→        <div className="h-1.5 min-w-24 flex-1 overflow-hidden rounded-full bg-secondary" aria-hidden>
   124→          <div
   125→            className={cn("h-full rounded-full transition-[width] duration-500", dead ? "bg-[var(--status-danger-dot)]" : "bg-kraft-strong")}
   126→            style={{ width: `${dead ? 100 : progress}%` }}
   127→          />
   128→        </div>
   129→      </div>
   130→
   131→      {/* Full rail — xl and up */}
   132→      <ol className="hidden items-start xl:flex" aria-label="Order progress steps">
   133→        {flow.map((s, i) => {
   134→          const entry = latestByStatus.get(s);
   135→          const done = i < currentIdx || (terminal && entry !== undefined);
   136→          const current = i === currentIdx && !terminal;
   137→          return (
   138→            <li
   139→              key={s}
   140→              className="relative flex min-w-[96px] flex-1 basis-0 flex-col items-center gap-2.5 px-1 text-center"
   141→            >
   142→              {i < flow.length - 1 && (
   143→                <span
   144→                  aria-hidden
   145→                  className={cn(
   146→                    "absolute top-[7px] left-[calc(50%+10px)] h-px w-[calc(100%-20px)]",
   147→                    done || current ? "bg-foreground/25" : "bg-border",
   148→                  )}
   149→                />
   150→              )}
   151→              <span
   152→                className={cn(
   153→                  "relative z-10 flex size-[14px] shrink-0 items-center justify-center rounded-full",
   154→                  done
   155→                    ? "bg-foreground"
   156→                    : current
   157→                      ? "border-2 border-kraft-strong bg-kraft"
   158→                      : "border border-border bg-card",
   159→                )}
   160→              >
   161→                {done && <span className="size-[4.5px] rounded-full bg-background" aria-hidden />}
   162→              </span>
   163→              <div className="min-w-0 max-w-full">
   164→                <p
   165→                  className={cn(
   166→                    "text-caption max-w-full truncate whitespace-nowrap",
   167→                    current
   168→                      ? "font-semibold text-foreground"
   169→                      : done
   170→                        ? "font-medium text-foreground/80"
   171→                        : "font-medium text-muted-foreground",
   172→                  )}
   173→                >
   174→                  {ORDER_STATUS_LABELS[s]}
   175→                </p>
   176→                {entry && (
   177→                  <p className="tnum text-caption whitespace-nowrap text-muted-foreground">
   178→                    {formatDate(entry.createdAt)}
   179→                  </p>
   180→                )}
   181→              </div>
   182→            </li>
   183→          );
   184→        })}
   185→      </ol>
   186→    </>
   187→  );
   188→}

   678→                <p>Any items currently in your cart will be replaced.</p>
   679→              </div>

   706→        </AlertDialogContent>
   707→      </AlertDialog>
   708→
   709→      {/* Unavailable items notice */}
   710→      <Dialog open={unavailableDraft !== null} onOpenChange={(o) => !o && setUnavailableDraft(null)}>
   711→        <DialogContent className="sm:max-w-[520px]">
   712→          <DialogHeader>
   713→            <DialogTitle className="flex items-center gap-2">
   714→              <TriangleAlert className="size-4.5 text-kraft-ink" aria-hidden />
   715→              Some items unavailable
   716→            </DialogTitle>
   717→            <DialogDescription>
   718→              These products from {unavailableDraft?.sourceOrderNumber} are inactive or not approved for
   719→              your account. Available items load if you continue.
   720→            </DialogDescription>
   721→          </DialogHeader>
   722→          <ul className="max-h-56 overflow-y-auto overscroll-contain rounded-[10px] border divide-y">
   723→            {(unavailableDraft?.lines ?? [])
   724→              .filter((l) => l.unavailable)
   725→              .map((l) => (
   726→                <li key={l.productId} className="flex items-center justify-between gap-3 px-3.5 py-2.5 text-body-sm">
   727→                  <span className="min-w-0">
   728→                    <span className="block truncate font-medium">{l.name}</span>
   729→                    <Mono className="text-muted-foreground">{l.sku}</Mono>
   730→                  </span>
   731→                  <span className="tnum shrink-0 text-caption text-muted-foreground">
   732→                    qty {l.qty.toLocaleString()}
   733→                  </span>
   734→                </li>
   735→              ))}
   736→          </ul>
   737→          <DialogFooter className="gap-2">
   738→            <Button variant="outline" onClick={() => setUnavailableDraft(null)}>
   739→              Cancel
   740→            </Button>
   741→            <Button
   742→              disabled={(unavailableDraft?.lines ?? []).every((l) => l.unavailable)}
   743→              onClick={() => {
   744→                if (unavailableDraft) finishReorder(unavailableDraft);
   745→                setUnavailableDraft(null);
   746→              }}

   748→              <Check className="size-4" aria-hidden />
   749→              Continue
   750→            </Button>
   751→          </DialogFooter>
   752→        </DialogContent>
   753→      </Dialog>
   754→    </motion.div>
   755→  );

   791→function ChangeRequestsCard({
   792→  orderId,
   793→  requests,
   794→}: {
   795→  orderId: string;
   796→  requests: ChangeRequestDto[];
   797→}) {
   798→  const queryClient = useQueryClient();
   799→
   800→  const respondMutation = useMutation({
   801→    mutationFn: (vars: { id: string; action: RespondAction }) =>
   802→      api.post<ChangeRequestDto>(`/api/customer/change-requests/${vars.id}/respond`, {
   803→        action: vars.action,
   804→      }),
   805→    onSuccess: (_cr, vars) => {
   806→      queryClient.invalidateQueries({ queryKey: ["orders"] });
   807→      queryClient.invalidateQueries({ queryKey: ["orders", "detail", orderId] });
   808→      queryClient.invalidateQueries({ queryKey: ["customer-dashboard"] });
   809→      queryClient.invalidateQueries({ queryKey: ["notifications"] });
   810→      if (vars.action === "ACCEPT") {
   811→        toast({ title: "Proposal accepted" });
   812→      } else if (vars.action === "DECLINE") {
   813→        toast({ title: "Proposal declined" });
   814→      } else {
   815→        toast({ title: "Request withdrawn" });
   816→      }
   817→    },
   818→    onError: (err) =>
   819→      toast({
   820→        title: "Could not respond",
   821→        description: errorMessage(err, "Please try again."),
   822→        variant: "destructive",
   823→      }),
   824→  });
   825→
   826→  // Newest-first from the API — render chronologically for the narrative.
   827→  const chronological = [...requests].reverse();
   828→
   829→  return (
   830→    <SectionCard
   831→      title="Change requests"
   832→      className="print:break-inside-avoid"
   833→    >
   834→      <div className="space-y-3">
   835→        {chronological.map((cr) => (
   836→          <ChangeRequestEntry
   837→            key={cr.id}
   838→            cr={cr}
   839→            busy={respondMutation.isPending}
   840→            onRespond={(id, action) => respondMutation.mutate({ id, action })}
   841→          />
   842→        ))}
   843→      </div>
   844→    </SectionCard>
   845→  );
   846→}
