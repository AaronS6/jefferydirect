   476→function ProductDialog({
   477→  categories,
   478→  product,
   479→  open,
   480→  onOpenChange,
   481→}: {
   482→  categories: CategoryDto[];
   483→  product?: AdminProduct;
   484→  open: boolean;
   485→  onOpenChange: (v: boolean) => void;
   486→}) {
   487→  const queryClient = useQueryClient();
   488→  const isEdit = Boolean(product);
   489→
   490→  const [form, setForm] = useState<ProductForm>(() =>
   491→    product
   492→      ? {
   493→          sku: product.sku,
   494→          name: product.name,
   495→          categoryId: product.categoryId,
   496→          shortDescription: product.shortDescription,
   497→          description: product.description,
   498→          material: product.material ?? "",
   499→          dimensions: product.dimensions ?? "",
   500→          colour: product.colour ?? "",
   501→          printingOptions: product.printingOptions ?? "",
   502→          casePack: String(product.casePack),
   503→          palletPack: product.palletPack != null ? String(product.palletPack) : "",
   504→          minOrderQty: String(product.minOrderQty),
   505→          increment: String(product.increment),
   506→          unit: product.unit,
   507→          defaultPrice: product.defaultPriceCents != null ? centsToInputValue(product.defaultPriceCents) : "",
   508→          stockStatus: product.stockStatus,
   509→          leadTimeNote: product.leadTimeNote ?? "",
   510→        }
   511→      : { ...EMPTY_PRODUCT_FORM },
   512→  );
   513→  const [submitError, setSubmitError] = useState<string | null>(null);
   514→
   515→  const parsedPrice = form.defaultPrice.trim() === "" ? null : parseDollarsToCents(form.defaultPrice);
   516→  const priceInvalid = form.defaultPrice.trim() !== "" && parsedPrice === null;
   517→
   518→  const save = useMutation<{ ok: boolean } | { id: string }, Error>({
   519→    mutationFn: async (): Promise<{ ok: boolean } | { id: string }> => {
   520→      const payload = {
   521→        sku: form.sku.trim(),
   522→        name: form.name.trim(),
   523→        categoryId: form.categoryId,
   524→        shortDescription: form.shortDescription.trim(),
   525→        description: form.description.trim(),
   526→        material: form.material.trim() || null,
   527→        dimensions: form.dimensions.trim() || null,
   528→        colour: form.colour.trim() || null,
   529→        printingOptions: form.printingOptions.trim() || null,
   530→        casePack: Number.parseInt(form.casePack, 10),
   531→        palletPack: form.palletPack.trim() === "" ? null : Number.parseInt(form.palletPack, 10),
   532→        minOrderQty: Number.parseInt(form.minOrderQty, 10),
   533→        increment: Number.parseInt(form.increment, 10),
   534→        unit: form.unit,
   535→        defaultPriceCents: form.defaultPrice.trim() === "" ? null : parsedPrice,
   536→        stockStatus: form.stockStatus,
   537→        leadTimeNote: form.leadTimeNote.trim() || null,
   538→      };
   539→      return isEdit
   540→        ? api.put<{ ok: boolean }>(`/api/admin/products/${product!.id}`, payload)
   541→        : api.post<{ id: string }>("/api/admin/products", payload);
   542→    },
   543→    onSuccess: () => {
   544→      queryClient.invalidateQueries({ queryKey: ["admin", "products"] });
   545→      queryClient.invalidateQueries({ queryKey: ["admin", "pricing"] });
   546→      queryClient.invalidateQueries({ queryKey: ["admin", "overview"] });
   547→      onOpenChange(false);
   548→      toast({
   549→        title: isEdit ? "Product updated" : "Product created",
   550→        description: `${form.sku.trim()} — ${form.name.trim()}`,
   551→      });
   552→    },
   553→    onError: (err: Error) => setSubmitError(err.message),
   554→  });
   555→
   556→  const set =
   557→    (key: keyof ProductForm) =>
   558→    (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) =>
   559→      setForm((f) => ({ ...f, [key]: e.target.value }));
   560→
   561→  function submit(e: React.FormEvent) {
   562→    e.preventDefault();
   563→    setSubmitError(null);
   564→    if (!form.sku.trim() || !form.name.trim() || !form.categoryId || !form.shortDescription.trim() || !form.description.trim()) {
   565→      setSubmitError("SKU, name, category and both descriptions are required.");
   566→      return;
   567→    }
   568→    for (const [label, raw] of [
   569→      ["Case pack", form.casePack],
   570→      ["Minimum order quantity", form.minOrderQty],
   571→      ["Quantity increment", form.increment],
   572→    ] as const) {
   573→      const n = Number.parseInt(raw, 10);
   574→      if (!Number.isFinite(n) || n < 1) {
   575→        setSubmitError(`${label} must be a whole number of 1 or more.`);
   576→        return;
   577→      }
   578→    }
   579→    if (form.palletPack.trim() !== "" && !(Number.parseInt(form.palletPack, 10) >= 1)) {
   580→      setSubmitError("Pallet pack must be a whole number of 1 or more.");
   581→      return;
   582→    }
   583→    if (priceInvalid) {
   584→      setSubmitError("Default price must be a valid dollar amount (e.g. 0.42), or blank for quote-required.");
   585→      return;
   586→    }
   587→    save.mutate();
   588→  }
   589→
   590→  return (
   591→    <Dialog open={open} onOpenChange={(v) => { onOpenChange(v); if (!v) setSubmitError(null); }}>
   592→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-[640px]">
   593→        <form onSubmit={submit} className="space-y-5">
   594→          <DialogHeader className="border-b pb-4">
   595→            <DialogTitle className="text-[15px] font-medium">
   596→              {isEdit ? `Edit product — ${product!.sku}` : "New catalogue product"}
   597→            </DialogTitle>
   598→            <DialogDescription className="text-[12.5px] leading-relaxed">
   599→              {isEdit
   600→                ? "Changes apply immediately across the catalogue. Blank default price means quote-required."
   601→                : "Create a product customers can be approved to order. SKU must be unique."}
   602→            </DialogDescription>
   603→          </DialogHeader>
   604→
   605→          <div className="grid gap-3.5 sm:grid-cols-2">
   606→            <div className="space-y-1.5">
   607→              <Label htmlFor="prod-sku" className="text-[12.5px]">SKU</Label>
   608→              <Input id="prod-sku" value={form.sku} onChange={set("sku")} placeholder="JG-CB-MAI4" className="h-9 font-mono text-[12.5px]" required />
   609→            </div>
   610→            <div className="space-y-1.5">
   611→              <Label htmlFor="prod-name" className="text-[12.5px]">Product name</Label>
   612→              <Input id="prod-name" value={form.name} onChange={set("name")} placeholder="White Mailer Carton 10 × 7 × 4 in" className="h-9 text-[13px]" required />
   613→            </div>
   614→            <div className="space-y-1.5 sm:col-span-2">
   615→              <Label htmlFor="prod-category" className="text-[12.5px]">Category</Label>
   616→              <Select value={form.categoryId} onValueChange={(v) => setForm((f) => ({ ...f, categoryId: v }))}>
   617→                <SelectTrigger id="prod-category" className="h-9 w-full text-[13px]" aria-label="Product category">
   618→                  <SelectValue placeholder="Choose a category…" />
   619→                </SelectTrigger>
   620→                <SelectContent>
   621→                  {categories.map((c) => (
   622→                    <SelectItem key={c.id} value={c.id}>
   623→                      {c.name}
   624→                    </SelectItem>
   625→                  ))}
   626→                </SelectContent>
   627→              </Select>
   628→            </div>
   629→            <div className="space-y-1.5 sm:col-span-2">
   630→              <Label htmlFor="prod-short" className="text-[12.5px]">Short description <span className="text-muted-foreground">(catalogue card)</span></Label>
   631→              <Input id="prod-short" value={form.shortDescription} onChange={set("shortDescription")} placeholder="Standard 10-inch pizza box, lock-corner, kraft." className="h-9 text-[13px]" required />
   632→            </div>
   633→            <div className="space-y-1.5 sm:col-span-2">
   634→              <Label htmlFor="prod-desc" className="text-[12.5px]">Full description</Label>
   635→              <Textarea id="prod-desc" rows={3} value={form.description} onChange={set("description")} placeholder="Construction, performance and packaging details…" className="text-[13px]" required />
   636→            </div>
   637→
   638→            <GroupLabel className="sm:col-span-2">Specifications</GroupLabel>
   639→            <div className="space-y-1.5">
   640→              <Label htmlFor="prod-dimensions" className="text-[12.5px]">Dimensions</Label>
   641→              <Input id="prod-dimensions" value={form.dimensions} onChange={set("dimensions")} placeholder="10 × 10 × 1.75 in" className="h-9 text-[13px]" />
   642→            </div>
   643→            <div className="space-y-1.5">
   644→              <Label htmlFor="prod-material" className="text-[12.5px]">Material</Label>
   645→              <Input id="prod-material" value={form.material} onChange={set("material")} placeholder="Single-wall corrugated kraft" className="h-9 text-[13px]" />
   646→            </div>
   647→            <div className="space-y-1.5">
   648→              <Label htmlFor="prod-colour" className="text-[12.5px]">Colour</Label>
   649→              <Input id="prod-colour" value={form.colour} onChange={set("colour")} placeholder="Natural kraft" className="h-9 text-[13px]" />
   650→            </div>
   651→            <div className="space-y-1.5">
   652→              <Label htmlFor="prod-printing" className="text-[12.5px]">Printing options</Label>
   653→              <Input id="prod-printing" value={form.printingOptions} onChange={set("printingOptions")} placeholder="1–4 colour flexo on volume" className="h-9 text-[13px]" />
   654→            </div>
   655→
   656→            <GroupLabel className="sm:col-span-2">Ordering rules</GroupLabel>
   657→            <div className="space-y-1.5">
   658→              <Label htmlFor="prod-casepack" className="text-[12.5px]">Case pack</Label>
   659→              <Input id="prod-casepack" inputMode="numeric" value={form.casePack} onChange={set("casePack")} placeholder="100" className="tnum h-9 text-[13px]" required />
   660→            </div>
   661→            <div className="space-y-1.5">
   662→              <Label htmlFor="prod-palletpack" className="text-[12.5px]">Pallet pack <span className="text-muted-foreground">(optional)</span></Label>
   663→              <Input id="prod-palletpack" inputMode="numeric" value={form.palletPack} onChange={set("palletPack")} placeholder="1200" className="tnum h-9 text-[13px]" />
   664→            </div>
   665→            <div className="space-y-1.5">
   666→              <Label htmlFor="prod-moq" className="text-[12.5px]">Minimum order qty</Label>
   667→              <Input id="prod-moq" inputMode="numeric" value={form.minOrderQty} onChange={set("minOrderQty")} placeholder="500" className="tnum h-9 text-[13px]" required />
   668→            </div>
   669→            <div className="space-y-1.5">
   670→              <Label htmlFor="prod-increment" className="text-[12.5px]">Quantity increment</Label>
   671→              <Input id="prod-increment" inputMode="numeric" value={form.increment} onChange={set("increment")} placeholder="100" className="tnum h-9 text-[13px]" required />
   672→            </div>
   673→            <div className="space-y-1.5">
   674→              <Label htmlFor="prod-unit" className="text-[12.5px]">Sold per</Label>
   675→              <Select value={form.unit} onValueChange={(v) => setForm((f) => ({ ...f, unit: v }))}>
   676→                <SelectTrigger id="prod-unit" className="h-9 w-full text-[13px]" aria-label="Selling unit">
   677→                  <SelectValue />
   678→                </SelectTrigger>
   679→                <SelectContent>
   680→                  {UNITS.map((u) => (
   681→                    <SelectItem key={u} value={u}>
   682→                      {u === "each" ? "Each (per piece)" : "Case"}
   683→                    </SelectItem>
   684→                  ))}
   685→                </SelectContent>
   686→              </Select>
   687→            </div>
   688→            <div className="space-y-1.5">
   689→              <Label htmlFor="prod-stock" className="text-[12.5px]">Stock status</Label>
   690→              <Select value={form.stockStatus} onValueChange={(v) => setForm((f) => ({ ...f, stockStatus: v }))}>
   691→                <SelectTrigger id="prod-stock" className="h-9 w-full text-[13px]" aria-label="Stock status">
   692→                  <SelectValue />
   693→                </SelectTrigger>
   694→                <SelectContent>
   695→                  {STOCK_STATUSES.map((s) => (
   696→                    <SelectItem key={s} value={s}>
   697→                      {STOCK_STATUS_LABELS[s]}
   698→                    </SelectItem>
   699→                  ))}
   700→                </SelectContent>
   701→              </Select>
   702→            </div>
   703→
   704→            <GroupLabel className="sm:col-span-2">Pricing</GroupLabel>
   705→            <div className="space-y-1.5">
   706→              <Label htmlFor="prod-price" className="text-[12.5px]">
   707→                Default price <span className="text-muted-foreground">(blank = quotation required)</span>
   708→              </Label>
   709→              <div className="relative">
   710→                <span className="pointer-events-none absolute left-2.5 top-1/2 -translate-y-1/2 text-[12px] text-muted-foreground" aria-hidden>$</span>
   711→                <Input
   712→                  id="prod-price"
   713→                  inputMode="decimal"
   714→                  value={form.defaultPrice}
   715→                  onChange={set("defaultPrice")}
   716→                  placeholder="0.42"
   717→                  aria-invalid={priceInvalid || undefined}
   718→                  className="tnum h-9 pl-6 text-[13px]"
   719→                />
   720→              </div>
   721→              {priceInvalid ? <FieldError>Enter a valid amount, e.g. 0.42</FieldError> : null}
   722→            </div>
   723→            <div className="space-y-1.5">
   724→              <Label htmlFor="prod-lead" className="text-[12.5px]">Lead time note <span className="text-muted-foreground">(optional)</span></Label>
   725→              <Input id="prod-lead" value={form.leadTimeNote} onChange={set("leadTimeNote")} placeholder="2 weeks from artwork approval" className="h-9 text-[13px]" />
   726→            </div>
   727→          </div>
   728→
   729→          {submitError && <FieldError>{submitError}</FieldError>}
   730→
   731→          <DialogFooter className="border-t pt-4">
   732→            <Button type="button" variant="ghost" onClick={() => onOpenChange(false)}>Cancel</Button>
   733→            <Button type="submit" className="gap-1.5" disabled={save.isPending}>
   734→              {save.isPending ? "Saving…" : isEdit ? "Save product" : "Create product"}
   735→            </Button>
   736→          </DialogFooter>
   737→        </form>
   738→      </DialogContent>
   739→    </Dialog>
   740→  );
   741→}
