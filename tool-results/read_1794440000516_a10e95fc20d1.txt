   263→  function loadDraftInto(d: OrderDraftDto) {
   264→    const payload = d.payload;
   265→    const catalogueLines = payload.lines.filter((l): l is typeof l & { productId: string } => l.productId !== null);
   266→    const customs = payload.lines.filter((l) => l.productId === null);
   267→
   268→    // Catalogue lines go back into the persisted cart store (extras stripped).
   269→    loadDraft(
   270→      catalogueLines.map((l) => ({
   271→        productId: l.productId,
   272→        sku: l.sku,
   273→        name: l.name,
   274→        qty: l.qty,
   275→        unitPriceCents: l.unitPriceCents ?? 0,
   276→        minOrderQty: l.minOrderQty,
   277→        increment: l.increment,
   278→        casePack: l.casePack,
   279→        unit: l.unit,
   280→      })),
   281→    );
   282→
   283→    const nextExtras: Record<string, LineExtras> = {};
   284→    for (const l of catalogueLines) {
   285→      if (l.note || cleanSpecs(l.specs) || (l.attachments?.length ?? 0) > 0) {
   286→        nextExtras[l.productId] = {
   287→          specs: l.specs ?? {},
   288→          note: l.note ?? "",
   289→          attachments: l.attachments ?? [],
   290→        };
   291→      }
   292→    }
   293→    setExtras(nextExtras);
   294→    setCustomLines(
   295→      customs.map((c) => ({
   296→        localId: newLocalId(),
   297→        name: c.name,
   298→        description: c.customDescription ?? "",
   299→        qty: c.qty,
   300→        note: c.note ?? "",
   301→        specs: c.specs ?? {},
   302→        attachments: c.attachments ?? [],
   303→      })),
   304→    );
   305→    const approvedIds = new Set(approvedAddresses.map((a) => a.id));
   306→    setAddressId(
   307→      payload.delivery.addressId && approvedIds.has(payload.delivery.addressId)
   308→        ? payload.delivery.addressId
   309→        : "",
   310→    );
   311→    setOrderNotes(payload.delivery.notes ?? "");
   312→    setRequestedDeliveryDate((payload.delivery.requestedDeliveryDate ?? "").slice(0, 10));
   313→    setPoNumber(payload.reference.poNumber ?? "");
   314→    setSpecialInstructions(payload.reference.specialInstructions ?? "");
   315→    setOrderAttachments(payload.attachments ?? []);
   316→    setDraftId(d.id);
   317→    setCustomFormOpen(false);
   318→    setSubmitError(null);
   319→    setStep(1);
   320→  }

   467→  function goTo(next: number) {
   468→    if (next > 1 && totalLineCount === 0) {
   469→      toast({
   470→        title: "Add at least one item",
   471→        description: "Add a catalogue product or a custom product request to continue.",
   472→      });
   473→      return;
   474→    }
   475→    if (next > 1 && !validQtys) {
   476→      toast({
   477→        title: "Check your quantities",
   478→        description: "Quantities must respect each product's minimum and multiples.",
   479→      });
   480→      return;
   481→    }
   482→    setSubmitError(null);
   483→    setStep(next);
   484→  }

   486→  function submit(e: React.FormEvent) {
   487→    e.preventDefault();
   488→    // Enter in any input behaves like "Continue" until the final step.
   489→    if (step < 4) {
   490→      goTo(step + 1);
   491→      return;
   492→    }
   493→    if (totalLineCount === 0) {
   494→      setSubmitError("Add at least one item before submitting.");
   495→      return;
   496→    }
   497→    if (!validQtys) {
   498→      setSubmitError("Quantities must respect each product's minimum and multiples.");
   499→      return;
   500→    }
   501→    setSubmitError(null);
   502→    submitMutation.mutate();
   503→  }

   515→      return;
   516→    }
   517→    setAddressFormError(null);
   518→    newAddressMutation.mutate();
   519→  }

   556→          }
   557→        />
   558→        {(drafts?.length ?? 0) > 0 && (
   559→          <div className="mt-8">
