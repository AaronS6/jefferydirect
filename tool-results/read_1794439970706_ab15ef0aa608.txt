    79→// Order progress hero — the full nine-step rail at xl and up; a compact
    80→// progress bar below xl (the vertical timeline below carries full history).
    81→// Ink dots for completed steps, kraft ring on the current one, hollow future.
    82→function StatusStepper({ status, history }: { status: string; history: OrderHistoryEntry[] }) {
    83→  const flow = ORDER_STATUSES;
    84→  const terminal = ORDER_TERMINAL_STATUSES.includes(status);
    85→  const currentIdx = flow.indexOf(status as (typeof flow)[number]);
    86→
    87→  // Latest history entry per status (ascending order).
    88→  const latestByStatus = new Map<string, OrderHistoryEntry>();
    89→  for (const h of [...history].sort((a, b) => a.createdAt.localeCompare(b.createdAt))) {
    90→    latestByStatus.set(h.status, h);
    91→  }
    92→
    93→  const progress =
    94→    currentIdx >= 0 ? Math.round(((currentIdx + 1) / flow.length) * 100) : 100;
    95→  const dead = status === "CANCELLED" || status === "REJECTED";
    96→
    97→  return (
    98→    <>
    99→      {/* Compact progress — below xl */}
   100→      <div className="flex flex-wrap items-center gap-x-4 gap-y-2.5 xl:hidden" aria-label="Order progress">
   101→        <span
   102→          aria-hidden
   103→          className={cn(
   104→            "flex size-[14px] shrink-0 items-center justify-center rounded-full",
   105→            dead
   106→              ? "border-2 border-[var(--status-danger-dot)] bg-[var(--status-danger-bg)]"
   107→              : terminal
   108→                ? "bg-foreground"
   109→                : "border-2 border-kraft-strong bg-kraft",
   110→          )}
   111→        >
   112→          {!dead && !terminal && <span className="size-1.5 rounded-full bg-background" aria-hidden />}
   113→          {terminal && <span className="size-[4.5px] rounded-full bg-background" aria-hidden />}
   114→        </span>
   115→        <p className="text-body font-semibold text-foreground">
   116→          {ORDER_STATUS_LABELS[status] ?? status}
   117→        </p>
   118→        {!dead && (
   119→          <p className="text-caption text-muted-foreground">
   120→            <span className="tnum">Step {Math.max(1, currentIdx + 1)} of {flow.length}</span>
   121→          </p>
   122→        )}
   123→        <div className="h-1.5 min-w-24 flex-1 overflow-hidden rounded-full bg-secondary" aria-hidden>
   124→          <div
   125→            className={cn("h-full rounded-full transition-[width] duration-500", dead ? "bg-[var(--status-danger-dot)]" : "bg-kraft-strong")}
   126→            style={{ width: `${dead ? 100 : progress}%` }}
   127→          />
   128→        </div>
   129→      </div>
   130→
   131→      {/* Full rail — xl and up */}
   132→      <ol className="hidden items-start xl:flex" aria-label="Order progress steps">
   133→        {flow.map((s, i) => {
   134→          const entry = latestByStatus.get(s);
   135→          const done = i < currentIdx || (terminal && entry !== undefined);
   136→          const current = i === currentIdx && !terminal;
   137→          return (
   138→            <li
   139→              key={s}
   140→              className="relative flex min-w-[96px] flex-1 basis-0 flex-col items-center gap-2.5 px-1 text-center"
   141→            >
   142→              {i < flow.length - 1 && (
   143→                <span
   144→                  aria-hidden
   145→                  className={cn(
   146→                    "absolute top-[7px] left-[calc(50%+10px)] h-px w-[calc(100%-20px)]",
   147→                    done || current ? "bg-foreground/25" : "bg-border",
   148→                  )}
   149→                />
   150→              )}
   151→              <span
   152→                className={cn(
   153→                  "relative z-10 flex size-[14px] shrink-0 items-center justify-center rounded-full",
   154→                  done
   155→                    ? "bg-foreground"
   156→                    : current
   157→                      ? "border-2 border-kraft-strong bg-kraft"
   158→                      : "border border-border bg-card",
   159→                )}
   160→              >
   161→                {done && <span className="size-[4.5px] rounded-full bg-background" aria-hidden />}
   162→              </span>
   163→              <div className="min-w-0 max-w-full">
   164→                <p
   165→                  className={cn(
   166→                    "text-caption max-w-full truncate whitespace-nowrap",
   167→                    current
   168→                      ? "font-semibold text-foreground"
   169→                      : done
   170→                        ? "font-medium text-foreground/80"
   171→                        : "font-medium text-muted-foreground",
   172→                  )}
   173→                >
   174→                  {ORDER_STATUS_LABELS[s]}
   175→                </p>
   176→                {entry && (
   177→                  <p className="tnum text-caption whitespace-nowrap text-muted-foreground">
   178→                    {formatDate(entry.createdAt)}
   179→                  </p>
   180→                )}
   181→              </div>
   182→            </li>
   183→          );
   184→        })}
   185→      </ol>
   186→    </>
   187→  );
   188→}

   701→                  submitting.

   703→                <p>Any items currently in your cart will be replaced.</p>
   704→              </div>
   705→            </AlertDialogDescription>

   707→          {reorderError && (
   708→            <div
   709→              role="alert"
   710→              className="rounded-[10px] bg-[var(--status-danger-bg)] px-3.5 py-2.5 text-body-sm text-[var(--status-danger-fg)]"

   737→          <DialogHeader>
   738→            <DialogTitle className="flex items-center gap-2">
   739→              <TriangleAlert className="size-4.5 text-kraft-ink" aria-hidden />
   740→              Some items are no longer available
   741→            </DialogTitle>
   742→            <DialogDescription>
   743→              The following products from {unavailableDraft?.sourceOrderNumber} are inactive or no longer
   744→              approved for your account — they can&apos;t be reordered. Available items will be loaded if you
   745→              continue.
   746→            </DialogDescription>
   747→          </DialogHeader>
   748→          <ul className="max-h-56 overflow-y-auto overscroll-contain rounded-[10px] border divide-y">
   749→            {(unavailableDraft?.lines ?? [])
   750→              .filter((l) => l.unavailable)
   751→              .map((l) => (
   752→                <li key={l.productId} className="flex items-center justify-between gap-3 px-3.5 py-2.5 text-body-sm">
   753→                  <span className="min-w-0">
   754→                    <span className="block truncate font-medium">{l.name}</span>
   755→                    <Mono className="text-muted-foreground">{l.sku}</Mono>
   756→                  </span>
   757→                  <span className="tnum shrink-0 text-caption text-muted-foreground">
   758→                    qty {l.qty.toLocaleString()}
   759→                  </span>
   760→                </li>
   761→              ))}
   762→          </ul>
   763→          <DialogFooter className="gap-2">
   764→            <Button variant="outline" onClick={() => setUnavailableDraft(null)}>
   765→              Cancel reorder
   766→            </Button>
   767→            <Button
   768→              disabled={(unavailableDraft?.lines ?? []).every((l) => l.unavailable)}
   769→              onClick={() => {
   770→                if (unavailableDraft) finishReorder(unavailableDraft);
   771→                setUnavailableDraft(null);
   772→              }}
   773→            >
   774→              <Check className="size-4" aria-hidden />
   775→              Continue with available items
   776→            </Button>
   777→          </DialogFooter>
   778→        </DialogContent>
   779→      </Dialog>
   780→    </motion.div>
   781→  );

   817→function ChangeRequestsCard({
   818→  orderId,
   819→  requests,
   820→}: {
   821→  orderId: string;
   822→  requests: ChangeRequestDto[];
   823→}) {
   824→  const queryClient = useQueryClient();
   825→
   826→  const respondMutation = useMutation({
   827→    mutationFn: (vars: { id: string; action: RespondAction }) =>
   828→      api.post<ChangeRequestDto>(`/api/customer/change-requests/${vars.id}/respond`, {
   829→        action: vars.action,
   830→      }),
   831→    onSuccess: (_cr, vars) => {
   832→      queryClient.invalidateQueries({ queryKey: ["orders"] });
   833→      queryClient.invalidateQueries({ queryKey: ["orders", "detail", orderId] });
   834→      queryClient.invalidateQueries({ queryKey: ["customer-dashboard"] });
   835→      queryClient.invalidateQueries({ queryKey: ["notifications"] });
   836→      if (vars.action === "ACCEPT") {
   837→        toast({
   838→          title: "Proposal accepted",
   839→          description: "Jeffery Group will apply it to your order — you'll see the change reflected here.",
   840→        });
   841→      } else if (vars.action === "DECLINE") {
   842→        toast({
   843→          title: "Proposal declined",
   844→          description: "The original order remains unchanged.",
   845→        });
   846→      } else {
   847→        toast({
   848→          title: "Request withdrawn",
   849→          description: "Your order continues exactly as it was.",
   850→        });
   851→      }
   852→    },
   853→    onError: (err) =>
   854→      toast({
   855→        title: "We could not record your response",
   856→        description: errorMessage(err, "Please try again."),
   857→        variant: "destructive",
   858→      }),
   859→  });
   860→
   861→  // Newest-first from the API — render chronologically for the narrative.
   862→  const chronological = [...requests].reverse();
   863→
   864→  return (
   865→    <SectionCard
   866→      title="Change requests"
   867→      description="Requests from your team on this order — nothing changes until Jeffery Group approves them."
   868→      className="print:break-inside-avoid"
   869→    >
   870→      <div className="space-y-3">
   871→        {chronological.map((cr) => (
   872→          <ChangeRequestEntry
   873→            key={cr.id}
   874→            cr={cr}
   875→            busy={respondMutation.isPending}
   876→            onRespond={(id, action) => respondMutation.mutate({ id, action })}
   877→          />
   878→        ))}
   879→      </div>
   880→    </SectionCard>
   881→  );
   882→}
