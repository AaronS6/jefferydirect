    36→    36→export function parseNotifyPrefs(json: string | null | undefined): Record<string, boolean> {
    37→    37→  try {
    38→    38→    const parsed: unknown = JSON.parse(json ?? "{}");
    39→    39→    if (parsed !== null && typeof parsed === "object" && !Array.isArray(parsed)) {
    40→    40→      const out: Record<string, boolean> = {};
    41→    41→      for (const [k, v] of Object.entries(parsed as Record<string, unknown>)) {
    42→    42→        if (typeof v === "boolean") out[k] = v;
    43→    43→      }
    44→    44→      return out;
    45→    45→    }
    46→    46→  } catch {
    47→    47→    // fall through to empty prefs
    48→    48→  }
    49→    49→  return {};
    50→    50→}

    52→    52→export function parseJson<T>(json: string | null | undefined, fallback: T): T {
    53→    53→  try {
    54→    54→    const parsed: unknown = JSON.parse(json ?? "null");
    55→    55→    if (parsed !== null && typeof parsed === "object") return parsed as T;
    56→    56→  } catch {
    57→    57→    // fall through to fallback
    58→    58→  }
    59→    59→  return fallback;
    60→    60→}

    62→    62→export function parseSpecJson(json: string | null | undefined): LineSpecs | null {
    63→    63→  if (!json) return null;
    64→    64→  try {
    65→    65→    const parsed: unknown = JSON.parse(json);
    66→    66→    if (parsed !== null && typeof parsed === "object" && !Array.isArray(parsed)) {
    67→    67→      return parsed as LineSpecs;
    68→    68→    }
    69→    69→  } catch {
    70→    70→    // fall through
    71→    71→  }
    72→    72→  return null;
    73→    73→}

    75→    75→export function parseAttachmentsJson(json: string | null | undefined): AttachmentMeta[] {
    76→    76→  if (!json) return [];
    77→    77→  try {
    78→    78→    const parsed: unknown = JSON.parse(json);
    79→    79→    if (Array.isArray(parsed)) {
    80→    80→      return parsed
    81→    81→        .filter(
    82→    82→          (a): a is AttachmentMeta =>
    83→    83→            a !== null && typeof a === "object" && typeof (a as AttachmentMeta).name === "string",
    84→    84→        )
    85→    85→        .map((a) => ({
    86→    86→          name: a.name,
    87→    87→          sizeLabel: typeof a.sizeLabel === "string" ? a.sizeLabel : "—",
    88→    88→          ...(typeof a.note === "string" ? { note: a.note } : {}),
    89→    89→        }));
    90→    90→    }
    91→    91→  } catch {
    92→    92→    // fall through
    93→    93→  }
    94→    94→  return [];
    95→    95→}

   510→   510→      orderNumber: o.orderNumber,
   511→   511→    }))
   512→   512→    .sort((a, b) => (a.createdAt < b.createdAt ? 1 : -1));
   513→   513→  const updates = [...o.updates]
   514→   514→    .sort((a, b) => {
   515→   515→      const at = (u: OrderUpdateRow) => (u.publishedAt ?? u.createdAt).getTime();
   516→   516→      return at(b) - at(a);
   517→   517→    })
   518→   518→    .map(serializeOrderUpdate);
   519→   519→  const changeRequests = [...o.changeRequests]
   520→   520→    .sort((a, b) => b.createdAt.getTime() - a.createdAt.getTime())
   521→   521→    .map((c) => serializeChangeRequest({ ...c, order: { id: o.id, orderNumber: o.orderNumber } }));
   522→   522→  return {
   523→   523→    id: o.id,
   524→   524→    orderNumber: o.orderNumber,
   525→   525→    status: o.status,
   526→   526→    orgId: o.orgId,
   527→   527→    orgName: o.organization.name,
   528→   528→    orgCode: o.organization.code,
   529→   529→    submittedBy: o.user.name,
   530→   530→    createdAt: o.createdAt.toISOString(),
   531→   531→    updatedAt: o.updatedAt.toISOString(),
   532→   532→    items,
   533→   533→    subtotalCents: o.subtotalCents,
   534→   534→    taxCents: o.taxCents,
   535→   535→    taxRateBp: o.taxRateBp,
   536→   536→    totalCents: o.totalCents,
   537→   537→    shippingText: o.shippingText,
   538→   538→    poNumber: o.poNumber,
   539→   539→    notes: o.notes,
   540→   540→    requestedDelivery: o.requestedDelivery,
   541→   541→    estimatedDelivery: o.estimatedDelivery ? o.estimatedDelivery.toISOString() : null,
   542→   542→    trackingNumber: o.trackingNumber,
   543→   543→    reorderSource: o.reorderSource,
   544→   544→    requestedDeliveryDate: o.requestedDeliveryDate ? o.requestedDeliveryDate.toISOString() : null,
   545→   545→    estimatedDispatch: o.estimatedDispatch ? o.estimatedDispatch.toISOString() : null,
   546→   546→    revisedDeliveryDate: o.revisedDeliveryDate ? o.revisedDeliveryDate.toISOString() : null,
   547→   547→    nextUpdateDate: o.nextUpdateDate ? o.nextUpdateDate.toISOString() : null,
   548→   548→    actualDeliveryDate: o.actualDeliveryDate ? o.actualDeliveryDate.toISOString() : null,
   549→   549→    sourceType: o.sourceType,
   550→   550→    recordedByName: o.recordedByName,
   551→   551→    history,
   552→   552→    documents,
   553→   553→    updates,
   554→   554→    changeRequests,
   555→   555→    ...(opts?.includeInternal
   556→   556→      ? { internalNotes: [...(o.internalNotes ?? [])].map(serializeInternalNote) }
   557→   557→      : {}),
   558→   558→  };
   559→   559→}
