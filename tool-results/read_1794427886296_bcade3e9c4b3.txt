   471→function UserRow({
   472→  user,
   473→  customerId,
   474→  onDone,
   475→}: {
   476→  user: CustomerProfile["users"][number];
   477→  customerId: string;
   478→  onDone: () => void;
   479→}) {
   480→  const queryClient = useQueryClient();
   481→  const [confirmDisable, setConfirmDisable] = useState(false);
   482→
   483→  const toggle = useMutation({
   484→    mutationFn: (status: "ACTIVE" | "DISABLED") =>
   485→      api.put<{ ok: boolean }>(`/api/admin/customers/${customerId}/users/${user.id}`, { status }),
   486→    onSuccess: (_res, status) => {
   487→      queryClient.invalidateQueries({ queryKey: ["admin", "customer", customerId] });
   488→      queryClient.invalidateQueries({ queryKey: ["admin", "customers"] });
   489→      onDone();
   490→      toast({
   491→        title: status === "ACTIVE" ? `${user.name} can sign in again` : `${user.name} disabled`,
   492→        description: status === "DISABLED" ? "They can no longer access the customer portal." : "Portal access restored.",
   493→      });
   494→      setConfirmDisable(false);
   495→    },
   496→    onError: (err: Error) => toast({ title: "Could not change user status.", description: err.message }),
   497→  });
   498→

   724→function InviteUserDialog({
   725→  open,
   726→  onOpenChange,
   727→  customerId,
   728→  onSaved,
   729→}: {
   730→  open: boolean;
   731→  onOpenChange: (v: boolean) => void;
   732→  customerId: string;
   733→  onSaved: () => void;
   734→}) {
   735→  const [form, setForm] = useState({ email: "", name: "", title: "" });
   736→  const [result, setResult] = useState<{ tempPassword: string } | null>(null);
   737→  const [submitError, setSubmitError] = useState<string | null>(null);
   738→
   739→  const invite = useMutation({
   740→    mutationFn: () =>
   741→      api.post<{ tempPassword: string }>(`/api/admin/customers/${customerId}/users`, {
   742→        email: form.email.trim(),
   743→        name: form.name.trim(),
   744→        title: form.title.trim() || undefined,
   745→      }),
   746→    onSuccess: (res) => {
   747→      setResult(res);
   748→      onSaved();
   749→      toast({ title: "Invitation created", description: "Share the temporary password to complete onboarding." });
   750→    },
   751→    onError: (err: Error) => setSubmitError(err.message),
   752→  });
   753→
   754→  function reset() {
   755→    setForm({ email: "", name: "", title: "" });
   756→    setResult(null);
   757→    setSubmitError(null);
   758→    invite.reset();
   759→  }
   760→
   761→  return (
   762→    <Dialog open={open} onOpenChange={(v) => { onOpenChange(v); if (!v) reset(); }}>
   763→      <DialogContent className="sm:max-w-[460px]">
   764→        {result ? (
   765→          <>
   766→            <DialogHeader className="border-b pb-4">
   767→              <DialogTitle className="text-[15px] font-medium">Portal user invited</DialogTitle>
   768→              <DialogDescription className="text-[12.5px] leading-relaxed">

   772→            </DialogHeader>
   773→            <TempPasswordPanel email={form.email.trim()} password={result.tempPassword} />
   774→            <DialogFooter className="border-t pt-4">
   775→              <Button variant="outline" onClick={reset}>Invite another</Button>
   776→              <Button onClick={() => { onOpenChange(false); reset(); }}>Done</Button>
   777→            </DialogFooter>
   778→          </>
   779→        ) : (
   780→          <form
   781→            onSubmit={(e) => {
   782→              e.preventDefault();
   783→              setSubmitError(null);
   784→              if (!form.email.trim() || !form.name.trim()) {
   785→                setSubmitError("Name and email are required.");
   786→                return;
   787→              }
   788→              if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email.trim())) {
   789→                setSubmitError("Enter a valid email address.");
   790→                return;
   791→              }
   792→              invite.mutate();
   793→            }}
   794→            className="space-y-4"
   795→          >
   796→            <DialogHeader className="border-b pb-4">
   797→              <DialogTitle className="text-[15px] font-medium">Invite portal user</DialogTitle>
   798→              <DialogDescription className="text-[12.5px] leading-relaxed">
   799→                Creates a user who can sign in and transact on behalf of this customer.
   800→              </DialogDescription>
   801→            </DialogHeader>
   802→            <div className="grid gap-3.5">
   803→              <div className="space-y-1.5">
   804→                <Label htmlFor="invite-name" className="text-[12.5px]">Full name</Label>
   805→                <Input id="invite-name" value={form.name} onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))} placeholder="e.g. Sofia Lindqvist" className="h-9 text-[13px]" required />
   806→              </div>

   808→                <Label htmlFor="invite-email" className="text-[12.5px]">Email address</Label>

   810→              </div>
   811→              <div className="space-y-1.5">
   812→                <Label htmlFor="invite-title" className="text-[12.5px]">Job title <span className="text-muted-foreground">(optional)</span></Label>
   813→                <Input id="invite-title" value={form.title} onChange={(e) => setForm((f) => ({ ...f, title: e.target.value }))} placeholder="e.g. Operations Manager" className="h-9 text-[13px]" />
   814→              </div>
   815→            </div>
   816→            {submitError && <p className="text-[12.5px] text-red-600">{submitError}</p>}
   817→            <DialogFooter className="border-t pt-4">

   826→      </DialogContent>
   827→    </Dialog>
