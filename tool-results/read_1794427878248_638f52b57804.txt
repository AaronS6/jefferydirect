    36→export function parseNotifyPrefs(json: string | null | undefined): Record<string, boolean> {
    37→  try {
    38→    const parsed: unknown = JSON.parse(json ?? "{}");
    39→    if (parsed !== null && typeof parsed === "object" && !Array.isArray(parsed)) {
    40→      const out: Record<string, boolean> = {};
    41→      for (const [k, v] of Object.entries(parsed as Record<string, unknown>)) {
    42→        if (typeof v === "boolean") out[k] = v;
    43→      }
    44→      return out;
    45→    }
    46→  } catch {
    47→    // fall through to empty prefs
    48→  }
    49→  return {};
    50→}
    51→
    52→export function parseJson<T>(json: string | null | undefined, fallback: T): T {
    53→  try {
    54→    const parsed: unknown = JSON.parse(json ?? "null");
    55→    if (parsed !== null && typeof parsed === "object") return parsed as T;
    56→  } catch {
    57→    // fall through to fallback
    58→  }
    59→  return fallback;
    60→}
    61→
    62→export function parseSpecJson(json: string | null | undefined): LineSpecs | null {
    63→  if (!json) return null;
    64→  try {
    65→    const parsed: unknown = JSON.parse(json);
    66→    if (parsed !== null && typeof parsed === "object" && !Array.isArray(parsed)) {
    67→      return parsed as LineSpecs;
    68→    }
    69→  } catch {
    70→    // fall through
    71→  }
    72→  return null;
    73→}
    74→
    75→export function parseAttachmentsJson(json: string | null | undefined): AttachmentMeta[] {
    76→  if (!json) return [];
    77→  try {
    78→    const parsed: unknown = JSON.parse(json);
    79→    if (Array.isArray(parsed)) {
    80→      return parsed
    81→        .filter(
    82→          (a): a is AttachmentMeta =>
    83→            a !== null && typeof a === "object" && typeof (a as AttachmentMeta).name === "string",
    84→        )
    85→        .map((a) => ({
    86→          name: a.name,
    87→          sizeLabel: typeof a.sizeLabel === "string" ? a.sizeLabel : "—",
    88→          ...(typeof a.note === "string" ? { note: a.note } : {}),
    89→        }));
    90→    }
    91→  } catch {
    92→    // fall through
    93→  }
    94→  return [];
    95→}
    96→
    97→export function parseChangeItems(json: string | null | undefined): ChangeItem[] {
    98→  if (!json) return [];
    99→  try {
   100→    const parsed: unknown = JSON.parse(json);
   101→    if (Array.isArray(parsed)) return parsed.filter((c) => c !== null && typeof c === "object") as ChangeItem[];
   102→  } catch {
   103→    // fall through
   104→  }
   105→  return [];
   106→}

   115→  } catch {

   117→  }
   118→  return null;
   119→}

   253→export function orderItemSummary(items: { qty: number }[]): {
   254→  itemCount: number;
   255→  totalUnits: number;
   256→  itemSummary: string;
   257→} {
   258→  const itemCount = items.length;
   259→  const totalUnits = items.reduce((sum, i) => sum + i.qty, 0);
   260→  return {
   261→    itemCount,
   262→    totalUnits,
   263→    itemSummary: `${itemCount} product${itemCount === 1 ? "" : "s"} · ${totalUnits.toLocaleString("en-CA")} units`,
   264→  };
   265→}

   483→
   484→export function serializeOrderDetail(o: OrderDetailRow, opts?: { includeInternal?: boolean }): OrderDetail {
   485→  const items: OrderItemDto[] = o.items.map((i) => ({
   486→    id: i.id,
   487→    productId: i.productId,
   488→    sku: i.sku,
   489→    name: i.name,
   490→    qty: i.qty,
   491→    unitPriceCents: i.unitPriceCents,
   492→    lineTotalCents: i.lineTotalCents,
   493→    note: i.note,
   494→    specs: parseSpecJson(i.specJson),
   495→    attachments: parseAttachmentsJson(i.specJson),
   496→  }));
   497→  const history = [...o.history]
   498→    .sort((a, b) => a.createdAt.getTime() - b.createdAt.getTime())
   499→    .map((h) => ({
   500→      id: h.id,
   501→      status: h.status,
   502→      note: h.note,
   503→      byName: h.byName,
   504→      createdAt: h.createdAt.toISOString(),
   505→    }));
   506→  const documents = o.documents
   507→    .map((d) => ({
   508→      ...serializeDocument(d),
   509→      // Documents listed inside an order always reference that order.
   510→      orderNumber: o.orderNumber,
   511→    }))
   512→    .sort((a, b) => (a.createdAt < b.createdAt ? 1 : -1));
   513→  const updates = [...o.updates]
   514→    .sort((a, b) => {
   515→      const at = (u: OrderUpdateRow) => (u.publishedAt ?? u.createdAt).getTime();
   516→      return at(b) - at(a);
   517→    })
   518→    .map(serializeOrderUpdate);
   519→  const changeRequests = [...o.changeRequests]
   520→    .sort((a, b) => b.createdAt.getTime() - a.createdAt.getTime())
   521→    .map((c) => serializeChangeRequest({ ...c, order: { id: o.id, orderNumber: o.orderNumber } }));
   522→  return {
   523→    id: o.id,
   524→    orderNumber: o.orderNumber,
   525→    status: o.status,
   526→    orgId: o.orgId,
   527→    orgName: o.organization.name,
   528→    orgCode: o.organization.code,
   529→    submittedBy: o.user.name,
   530→    createdAt: o.createdAt.toISOString(),
   531→    updatedAt: o.updatedAt.toISOString(),
   532→    items,
   533→    subtotalCents: o.subtotalCents,
   534→    taxCents: o.taxCents,
   535→    taxRateBp: o.taxRateBp,
   536→    totalCents: o.totalCents,
   537→    shippingText: o.shippingText,
   538→    poNumber: o.poNumber,
   539→    notes: o.notes,
   540→    requestedDelivery: o.requestedDelivery,
   541→    estimatedDelivery: o.estimatedDelivery ? o.estimatedDelivery.toISOString() : null,
   542→    trackingNumber: o.trackingNumber,
   543→    reorderSource: o.reorderSource,
   544→    requestedDeliveryDate: o.requestedDeliveryDate ? o.requestedDeliveryDate.toISOString() : null,
   545→    estimatedDispatch: o.estimatedDispatch ? o.estimatedDispatch.toISOString() : null,
   546→    revisedDeliveryDate: o.revisedDeliveryDate ? o.revisedDeliveryDate.toISOString() : null,
   547→    nextUpdateDate: o.nextUpdateDate ? o.nextUpdateDate.toISOString() : null,
   548→    actualDeliveryDate: o.actualDeliveryDate ? o.actualDeliveryDate.toISOString() : null,
   549→    sourceType: o.sourceType,
   550→    recordedByName: o.recordedByName,
   551→    history,
   552→    documents,
   553→    updates,
   554→    changeRequests,
   555→    ...(opts?.includeInternal
   556→      ? { internalNotes: [...(o.internalNotes ?? [])].map(serializeInternalNote) }
   557→      : {}),
   558→  };
   559→}

   825→export function serializeDraft(d: OrderDraftRow): OrderDraftDto {
   826→  let payload: OrderDraftPayload = { lines: [], delivery: { addressId: null, requestedDeliveryDate: null, notes: null }, reference: { poNumber: null, specialInstructions: null }, attachments: [] };
   827→  try {
   828→    const parsed: unknown = JSON.parse(d.payload);
   829→    if (parsed !== null && typeof parsed === "object" && Array.isArray((parsed as OrderDraftPayload).lines)) {
   830→      payload = parsed as OrderDraftPayload;
   831→    }
   832→  } catch {
   833→    // keep empty payload
   834→  }
   835→  return {
   836→    id: d.id,
   837→    name: d.name,
   838→    payload,
   839→    lineCount: payload.lines.length,
   840→    createdAt: d.createdAt.toISOString(),
   841→    updatedAt: d.updatedAt.toISOString(),
   842→  };
   843→}
