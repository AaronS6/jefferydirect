   283→   283→function AddressesTab() {
   284→   284→  const queryClient = useQueryClient();
   285→   285→  const [dialogOpen, setDialogOpen] = useState(false);
   286→   286→  const [form, setForm] = useState(EMPTY_ADDRESS);
   287→   287→  const [formError, setFormError] = useState<string | null>(null);
   288→   288→  const [removing, setRemoving] = useState<AddressDto | null>(null);
   289→   289→
   290→   290→  const { data: addresses, isLoading, error, refetch } = useQuery({
   291→   291→    queryKey: ["customer-addresses"],
   292→   292→    queryFn: () => api.get<AddressDto[]>("/api/customer/addresses"),
   293→   293→  });
   294→   294→
   295→   295→  const invalidate = () => queryClient.invalidateQueries({ queryKey: ["customer-addresses"] });
   296→   296→
   297→   297→  const addMutation = useMutation({
   298→   298→    mutationFn: () =>
   299→   299→      api.post<AddressDto>("/api/customer/addresses", {
   300→   300→        label: form.label.trim(),
   301→   301→        line1: form.line1.trim(),
   302→   302→        line2: form.line2.trim() || null,
   303→   303→        city: form.city.trim(),
   304→   304→        province: form.province.trim(),
   305→   305→        postal: form.postal.trim(),
   306→   306→      }),
   307→   307→    onSuccess: (a) => {
   308→   308→      invalidate();
   309→   309→      setDialogOpen(false);
   310→   310→      setForm(EMPTY_ADDRESS);
   311→   311→      setFormError(null);
   312→   312→      toast({ title: "Address added" });
   313→   313→    },
   314→   314→    onError: (err) =>
   315→   315→      setFormError(err instanceof Error ? err.message : "We could not add the address. Please try again."),
   316→   316→  });
   317→   317→
   318→   318→  const setDefaultMutation = useMutation({
   319→   319→    mutationFn: (id: string) => api.patch<AddressDto>(`/api/customer/addresses/${id}`, { isDefault: true }),
   320→   320→    onSuccess: () => {
   321→   321→      invalidate();
   322→   322→      toast({ title: "Default address updated" });
   323→   323→    },
   324→   324→    onError: (err) =>
   325→   325→      toast({
   326→   326→        title: "Could not update address",
   327→   327→        description: err instanceof Error ? err.message : "Please try again.",
   328→   328→        variant: "destructive",
   329→   329→      }),
   330→   330→  });
   331→   331→
   332→   332→  const removeMutation = useMutation({
   333→   333→    mutationFn: (id: string) => api.del<{ ok: boolean }>(`/api/customer/addresses/${id}`),
   334→   334→    onSuccess: () => {
   335→   335→      invalidate();
   336→   336→      setRemoving(null);
   337→   337→      toast({ title: "Address removed" });
   338→   338→    },
   339→   339→    onError: (err) =>
   340→   340→      toast({
   341→   341→        title: "Could not remove address",
   342→   342→        description: err instanceof Error ? err.message : "Please try again.",
   343→   343→        variant: "destructive",
   344→   344→      }),
   345→   345→  });
   346→   346→
   347→   347→  function submitAddress(e: React.FormEvent) {
   348→   348→    e.preventDefault();
   349→   349→    if (!form.label.trim() || !form.line1.trim() || !form.city.trim() || !form.province.trim() || !form.postal.trim()) {
   350→   350→      setFormError("Label, street, city, province and postal code are required.");
   351→   351→      return;
   352→   352→    }
   353→   353→    setFormError(null);
   354→   354→    addMutation.mutate();
   355→   355→  }

   357→   357→  return (
   358→   358→    <SectionCard
   359→   359→      title="Delivery addresses"
   360→   360→      actions={

   362→   362→          variant="outline"
   363→   363→          size="sm"

   367→   367→            setFormError(null);
   368→   368→            setDialogOpen(true);

   376→   376→        <ErrorAlert message="We could not load your addresses." onRetry={() => refetch()} />
   377→   377→      ) : isLoading || addresses === undefined ? (
   378→   378→        <RowsSkeleton rows={3} />
   379→   379→      ) : addresses.length === 0 ? (
   380→   380→        <div className="flex items-start gap-2.5 rounded-[10px] border border-kraft/40 bg-kraft-soft/60 px-4 py-3 text-body-sm text-kraft-ink">
   381→   381→          <Building2 className="mt-0.5 size-4 shrink-0" aria-hidden />
   382→   382→          <p>No delivery addresses yet.</p>

   386→   386→          {addresses.map((a) => (

   390→   390→            >
   391→   391→              <div className="flex items-start justify-between gap-3">
   392→   392→                <div className="min-w-0">
   393→   393→                  <p className="flex flex-wrap items-center gap-2 text-body-sm font-semibold text-foreground">
   394→   394→                    {a.label}
   395→   395→                    {a.isDefault && (
   396→   396→                      <span className="inline-flex h-[22px] items-center rounded-full border bg-card px-2.5 text-caption font-medium text-muted-foreground">
   397→   397→                        Default
   398→   398→                      </span>
   399→   399→                    )}
   400→   400→                  </p>
   401→   401→                  <p className="mt-1.5 text-body-sm leading-relaxed text-muted-foreground">
   402→   402→                    {a.line1}
   403→   403→                    {a.line2 ? <>, {a.line2}</> : null}

   405→   405→                    {a.city}, {a.province} {a.postal}

   407→   407→                </div>
   408→   408→              </div>

   410→   410→                {!a.isDefault && (
   411→   411→                  <Button
   412→   412→                    variant="outline"
   413→   413→                    size="sm"
   414→   414→                    className="h-7 gap-1.5 px-2.5"
