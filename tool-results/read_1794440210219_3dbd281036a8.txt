    50→// ─── Builder draft state ─────────────────────────────────────────────────────
    51→
    52→type DraftRow = { uid: string; description: string; kind: string; qty: string; price: string };
    53→
    54→let uidCounter = 0;
    55→const newUid = () => `row-${++uidCounter}`;
    56→
    57→function rowsFromQuote(quote: QuotationDetail): DraftRow[] {
    58→  if (quote.items.length === 0) return [emptyRow()];
    59→  return quote.items.map((i) => ({
    60→    uid: newUid(),
    61→    description: i.description,
    62→    kind: i.kind,
    63→    qty: String(i.qty),
    64→    price: centsToInputValue(i.unitPriceCents),
    65→  }));
    66→}
    67→
    68→function emptyRow(): DraftRow {
    69→  return { uid: newUid(), description: "", kind: "PRODUCT", qty: "", price: "" };
    70→}
    71→
    72→function rowCents(row: DraftRow): number | null {
    73→  const cents = parseDollarsToCents(row.price.trim());
    74→  const qty = Number.parseInt(row.qty, 10);
    75→  if (cents === null || !Number.isFinite(qty) || qty <= 0) return null;
    76→  return cents * qty;
    77→}
    78→
    79→function rowIssues(row: DraftRow): { description: boolean; qty: boolean; price: boolean } {
    80→  const cents = parseDollarsToCents(row.price.trim());
    81→  const qty = Number.parseInt(row.qty, 10);
    82→  return {
    83→    description: row.description.trim() === "",
    84→    qty: !Number.isFinite(qty) || qty <= 0,
    85→    price: cents === null,
    86→  };
    87→}

   113→    if (!quote) return;

   115→      initialised.current = true;

   132→    mutationFn: () => api.post<{ ok: boolean }>(`/api/admin/quotations/${quotationId}/acknowledge`, { status: "UNDER_REVIEW" }),
   133→    onSuccess: () => {
   134→      invalidate();
   135→      toast({
   136→        title: "Request acknowledged",
   137→        description: "The customer has been notified their request is under review.",
   138→      });
   139→    },
   140→    onError: (err: Error) => toast({ title: "Could not acknowledge.", description: err.message }),

   147→        kind: r.kind,
   148→        qty: Number.parseInt(r.qty, 10),
   149→        unitPriceCents: parseDollarsToCents(r.price.trim()) ?? 0,
   150→      }));

   152→        items,
   153→        expiryDate: expiry || null,
   154→        terms: terms.trim() || null,
   155→        internalNote: internalNote.trim() || null,
   156→      });
   157→    },
   158→    onSuccess: () => {
   159→      invalidate();
   160→      setBuilderOpen(false);
   161→      initialised.current = false; // re-prefill from refreshed data if reopened
   162→      toast({
   163→        title: "Quotation issued — customer notified",
   164→        description: "The customer can now review pricing and respond from their portal.",
   165→      });
   166→    },
   167→    onError: (err: Error) => toast({ title: "Could not issue quotation.", description: err.message }),
   168→  });

   171→    return (
   172→      <div>
   173→        <PageHeader title="Quotation" onBack={back} eyebrow="Quotation detail" />
   174→        <InlineError message={error instanceof Error ? error.message : "Could not load this quotation."} />
   175→      </div>
   176→    );
   177→  }
   178→
   179→  if (isLoading || !quote) {
   180→    return (
   181→      <div>
   182→        <PageHeader title="Quotation" onBack={back} eyebrow="Quotation detail" description="Loading quotation…" />
   183→        <div className="grid gap-4 sm:gap-6 lg:grid-cols-[1fr_360px]">
   184→          <SectionCard>
   185→            <RowsSkeleton rows={5} />
   186→          </SectionCard>
   187→          <SectionCard>
   188→            <RowsSkeleton rows={4} />
   189→          </SectionCard>
   190→        </div>
   191→      </div>
   192→    );
   193→  }
   194→
   195→  const preIssue = quote.status === "REQUESTED" || quote.status === "UNDER_REVIEW";
   196→  const reissuable = quote.status === "ISSUED" || quote.status === "REVISED" || quote.status === "EXPIRED";
   197→  const responded = quote.status === "ACCEPTED" || quote.status === "REJECTED";
   198→  const hasItems = quote.items.length > 0;

   211→  });
   212→  const invalidRows = rows.length - rows.filter((r) => {
   213→    const i = rowIssues(r);
   214→    return !i.description && !i.qty && !i.price;

   224→        description={

   226→            <button
   227→              type="button"

   229→              onClick={() => navigate("admin-customer-detail", { id: quote.orgId })}
   230→            >
   231→              {quote.orgName}

   233→            <span className="text-muted-foreground/50">·</span>
   234→            <span>Requested by {quote.requestedBy}</span>

   236→              <>
   237→                <span className="text-muted-foreground/50">·</span>
   238→                <span className="text-kraft-ink">
   239→                  Issued on behalf ({(ORDER_SOURCE_SHORT_LABELS[quote.sourceType] ?? quote.sourceType).toLowerCase()})
   240→                  {quote.recordedByName ? ` — ${quote.recordedByName}` : ""}
   241→                </span>
   242→              </>
   243→            )}
   244→            <span className="text-muted-foreground/50">·</span>
   245→            <span>{formatDate(quote.createdAt)}</span>
   246→          </span>
   247→        }
   248→        actions={
   249→          <div className="flex flex-wrap items-center gap-2">
   250→            {quote.revision > 0 && (
   251→              <span className="rounded-full border bg-secondary px-2 py-0.5 text-caption font-medium text-muted-foreground">
   252→                Revision {quote.revision}
   253→              </span>
   254→            )}
   255→            {preIssue && (
   256→              <Button
   257→                size="sm"
   258→                variant="outline"
   259→                className="gap-1.5"
   260→                disabled={quote.status !== "REQUESTED" || acknowledge.isPending}
   261→                onClick={() => acknowledge.mutate()}
   262→              >
   263→                {acknowledge.isPending ? <Loader2 className="size-3.5 animate-spin" aria-hidden /> : <CheckCircle2 className="size-3.5" aria-hidden />}
   264→                Acknowledge
   265→              </Button>
   266→            )}
   267→            <StatusBadge status={quote.status} kind="quote" />
   268→          </div>
   269→        }
   270→      />
   271→

   273→      <div className="grid items-start gap-4 sm:gap-6 lg:grid-cols-[minmax(0,1fr)_360px]">
   274→        {/* Zone A + issued quote + builder */}
   275→        <div className="min-w-0 space-y-4 sm:space-y-6">
   276→          <SectionCard
   277→            title="Customer's request"
   278→            description="What the customer asked for, in their own words"

   280→            <dl>
   281→              <DefRow label="Category" value={dashed(quote.category)} />

   283→              <DefRow label="Dimensions" value={dashed(quote.dimensions)} />
   284→              <DefRow label="Material" value={dashed(quote.material)} />
   285→              <DefRow label="Colour" value={dashed(quote.colour)} />
   286→              <DefRow label="Printing" value={dashed(quote.printing)} />
   287→              <DefRow label="Quantity" value={dashed(quote.quantity)} />
   288→              <DefRow label="Needed by" value={dashed(quote.neededBy)} />
   289→              <DefRow label="Destination" value={dashed(quote.destination)} />
   290→              <DefRow label="Specifications" value={dashed(quote.specs)} />
   291→              <DefRow label="Comments" value={dashed(quote.comments)} />
   292→            </dl>
   293→          </SectionCard>
   294→
   295→          {/* Response zone */}
   296→          {responded && (
   297→            <div
   298→              className={cn(
   299→                "flex items-start gap-3 rounded-[14px] border border-transparent p-4 sm:p-5",
   300→                quote.status === "ACCEPTED"
   301→                  ? "bg-[var(--status-done-bg)]"
   302→                  : "bg-[var(--status-danger-bg)]",
   303→              )}
   304→            >
   305→              {quote.status === "ACCEPTED" ? (
   306→                <CheckCircle2 className="mt-0.5 size-5 shrink-0 text-[var(--status-done-fg)]" aria-hidden />
   307→              ) : (
   308→                <XCircle className="mt-0.5 size-5 shrink-0 text-[var(--status-danger-fg)]" aria-hidden />
   309→              )}
   310→              <div className="min-w-0">
   311→                <p
   312→                  className={cn(
   313→                    "text-body font-medium",
   314→                    quote.status === "ACCEPTED" ? "text-[var(--status-done-fg)]" : "text-[var(--status-danger-fg)]",
   315→                  )}
   316→                >
   317→                  Customer {quote.status === "ACCEPTED" ? "accepted" : "declined"} this quotation
   318→                  {quote.respondedAt ? ` · ${formatDateTime(quote.respondedAt)}` : ""}
   319→                </p>
   320→                {quote.respondedNote && (
   321→                  <p className="mt-1 text-body-sm leading-relaxed text-foreground/80">{quote.respondedNote}</p>
   322→                )}
   323→              </div>
   324→            </div>
   325→          )}
   326→
   327→          {quote.status === "EXPIRED" && hasItems && (
   328→            <div className="flex items-start gap-3 rounded-[14px] border bg-secondary/50 p-4 text-body-sm leading-relaxed text-muted-foreground">
   329→              <CalendarDays className="mt-0.5 size-4 shrink-0" aria-hidden />
   330→              <p>
   331→                This quotation expired{quote.expiryDate ? ` on ${formatDate(quote.expiryDate)}` : ""} without a customer
   332→                response. Re-issue to send fresh pricing.
   333→              </p>
   334→            </div>
   335→          )}
   336→
   337→          {/* Issued quotation view */}
   338→          {hasItems && (reissuable || responded || quote.status === "EXPIRED") && (
   339→            <SectionCard
   340→              title={quote.status === "REVISED" ? `Issued quotation — revision ${quote.revision}` : "Issued quotation"}
   341→              description={
   342→                quote.expiryDate ? `Issued ${formatDate(quote.issuedAt)} · valid until ${formatDate(quote.expiryDate)}` : `Issued ${formatDate(quote.issuedAt)}`
   343→              }
   344→              action={
   345→                <div className="flex flex-wrap gap-2">
   346→                  <Button variant="outline" size="sm" className="gap-1.5" onClick={() => setPreviewOpen(true)}>
   347→                    <Eye className="size-3.5" aria-hidden /> Preview document
   348→                  </Button>
   349→                  {reissuable && (
   350→                    <Button
   351→                      variant={preIssue ? "default" : "outline"}
   352→                      size="sm"
   353→                      className="gap-1.5"
   354→                      onClick={() => {
   355→                        setBuilderOpen(false);
   356→                        openBuilderPrefilled(quote);
   357→                      }}
   358→                    >
   359→                      <RefreshCcw className="size-3.5" aria-hidden /> Re-issue (revision)
   360→                    </Button>

   365→              <Table>
   366→                <TableHeader>
   367→                  <TableRow>
   368→                    <Th>Description</Th>
   369→                    <Th>Type</Th>
   370→                    <Th className="text-right">Qty</Th>
   371→                    <Th className="text-right">Unit</Th>
   372→                    <Th className="text-right">Amount</Th>
   373→                  </TableRow>
   374→                </TableHeader>
   375→                <TableBody>
   376→                  {quote.items.map((item) => (
   377→                    <TableRow key={item.id}>
   378→                      <Td className="max-w-[320px] whitespace-normal">{item.description}</Td>
   379→                      <Td>
   380→                        <span className="inline-flex items-center rounded-full border bg-card px-2 py-0.5 text-caption text-muted-foreground">
   381→                          {QUOTE_ITEM_KIND_LABELS[item.kind] ?? item.kind}

   386→                        <Money cents={item.unitPriceCents} />
