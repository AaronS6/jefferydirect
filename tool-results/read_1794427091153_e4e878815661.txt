     1→"use client";
     2→
     3→// Order cart — a four-step request flow in one page: review lines, delivery,
     4→// reference details, then submit. Copy is honest throughout: this is a REQUEST
     5→// that Jeffery Group reviews and confirms.
     6→
     7→import { useMemo, useState } from "react";
     8→import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
     9→import {
    10→  ArrowLeft,
    11→  ArrowRight,
    12→  Check,
    13→  ChevronRight,
    14→  Loader2,
    15→  ShoppingCart,
    16→  Trash2,
    17→  Truck,
    18→} from "lucide-react";
    19→import { Button } from "@/components/ui/button";
    20→import { Input } from "@/components/ui/input";
    21→import { Label } from "@/components/ui/label";
    22→import { Textarea } from "@/components/ui/textarea";
    23→import { Separator } from "@/components/ui/separator";
    24→import { Skeleton } from "@/components/ui/skeleton";
    25→import {
    26→  Select,
    27→  SelectContent,
    28→  SelectItem,
    29→  SelectTrigger,
    30→  SelectValue,
    31→} from "@/components/ui/select";
    32→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table";
    33→import { PageHeader } from "@/components/shared/page-header";
    34→import { Money, MoneyWithCode } from "@/components/shared/money";
    35→import { ProductArtwork } from "@/components/shared/product-artwork";
    36→import { EmptyState } from "@/components/shared/empty-state";
    37→import { api } from "@/lib/api-client";
    38→import { useAppStore, useCartStore } from "@/store/app-store";
    39→import type { CartLine } from "@/types";
    40→import { taxFromSubtotalCents } from "@/lib/money";
    41→import { DEFAULT_TAX_RATE_BP } from "@/lib/constants";
    42→import { toast } from "@/hooks/use-toast";
    43→import { cn } from "@/lib/utils";
    44→import { ErrorAlert, Mono, QtyStepper, SectionCard } from "./portal-bits";
    45→
    46→type AccountPayload = {
    47→  addresses: { id: string; label: string; line1: string; line2: string | null; city: string; province: string; postal: string; isDefault: boolean }[];
    48→};
    49→
    50→const STEPS = [
    51→  { n: 1, label: "Review" },
    52→  { n: 2, label: "Delivery" },
    53→  { n: 3, label: "Reference" },
    54→  { n: 4, label: "Submit" },
    55→] as const;
    56→
    57→export function CartView() {
    58→  const navigate = useAppStore((s) => s.navigate);
    59→  const lines = useCartStore((s) => s.lines);
    60→  const setQty = useCartStore((s) => s.setQty);
    61→  const removeLine = useCartStore((s) => s.removeLine);
    62→  const clear = useCartStore((s) => s.clear);
    63→
    64→  const queryClient = useQueryClient();
    65→  const [step, setStep] = useState(1);
    66→  const [addressId, setAddressId] = useState<string>("");
    67→  const [orderNotes, setOrderNotes] = useState("");
    68→  const [requestedDelivery, setRequestedDelivery] = useState("");
    69→  const [poNumber, setPoNumber] = useState("");
    70→  const [specialInstructions, setSpecialInstructions] = useState("");
    71→  const [submitError, setSubmitError] = useState<string | null>(null);
    72→
    73→  const { data: account, isLoading: accountLoading, error: accountError } = useQuery({
    74→    queryKey: ["customer-account"],
    75→    queryFn: () => api.get<AccountPayload>("/api/customer/account"),
    76→    enabled: lines.length > 0,
    77→  });
    78→
    79→  // Default the address selection once account data arrives.
    80→  const addresses = account?.addresses ?? [];
    81→  const selectedAddress =
    82→    addresses.find((a) => a.id === addressId) ??
    83→    (addressId === "" ? (addresses.find((a) => a.isDefault) ?? addresses[0]) : undefined);
    84→
    85→  const subtotal = lines.reduce((sum, l) => sum + l.qty * l.unitPriceCents, 0);
    86→  const tax = taxFromSubtotalCents(subtotal, DEFAULT_TAX_RATE_BP);
    87→  const total = subtotal + tax;
    88→
    89→  const validQtys = lines.every((l) => l.qty >= l.minOrderQty && l.qty % l.increment === 0);
    90→
    91→  const submitMutation = useMutation({
    92→    mutationFn: () => {
    93→      const notes = [
    94→        orderNotes.trim() ? `Order notes: ${orderNotes.trim()}` : null,
    95→        specialInstructions.trim() ? `Special instructions: ${specialInstructions.trim()}` : null,
    96→      ]
    97→        .filter(Boolean)
    98→        .join("\n\n");
    99→      const chosen = selectedAddress ?? undefined;
   100→      return api.post<{ id: string; orderNumber: string }>("/api/customer/orders", {
   101→        items: lines.map((l) => ({ productId: l.productId, qty: l.qty })),
   102→        ...(chosen ? { addressId: chosen.id } : {}),
   103→        poNumber: poNumber.trim() || undefined,
   104→        notes: notes || undefined,
   105→        requestedDelivery: requestedDelivery.trim() || undefined,
   106→      });
   107→    },
   108→    onSuccess: (res) => {
   109→      clear();
   110→      queryClient.invalidateQueries({ queryKey: ["orders"] });
   111→      queryClient.invalidateQueries({ queryKey: ["customer-dashboard"] });
   112→      queryClient.invalidateQueries({ queryKey: ["notifications"] });
   113→      toast({
   114→        title: "Order request submitted",
   115→        description: `${res.orderNumber} — we'll confirm availability and pricing, then notify you.`,
   116→      });
   117→      navigate("order-detail", { id: res.id });
   118→    },
   119→    onError: (err) =>
   120→      setSubmitError(
   121→        err instanceof Error ? err.message : "We could not submit your order request. Please try again.",
   122→      ),
   123→  });
   124→
   125→  function goTo(next: number) {
   126→    if (next > 1 && !validQtys) {
   127→      toast({
   128→        title: "Check your quantities",
   129→        description: "Quantities must respect each product's minimum and multiples.",
   130→      });
   131→      return;
   132→    }
   133→    setSubmitError(null);
   134→    setStep(next);
   135→  }
   136→
   137→  function submit(e: React.FormEvent) {
   138→    e.preventDefault();
   139→    // Enter in any input behaves like "Continue" until the final step.
   140→    if (step < 4) {
   141→      goTo(step + 1);
   142→      return;
   143→    }
   144→    if (!validQtys) {
   145→      setSubmitError("Quantities must respect each product's minimum and multiples.");
   146→      return;
   147→    }
   148→    setSubmitError(null);
   149→    submitMutation.mutate();
   150→  }
   151→
   152→  if (lines.length === 0) {
   153→    return (
   154→      <div>
   155→        <PageHeader title="Order Cart" description="Build an order request from your approved products." />
   156→        <EmptyState
   157→          icon={ShoppingCart}
   158→          title="Your cart is empty"
   159→          description="Browse your approved catalogue and add products — pricing specific to your account is shown on every item."
   160→          action={
   161→            <Button onClick={() => navigate("catalog")} className="gap-2">
   162→              Browse products <ChevronRight className="size-4" aria-hidden />
   163→            </Button>
   164→          }
   165→        />
   166→      </div>
   167→    );
   168→  }
   169→
   170→  return (
   171→    <div>
   172→      <PageHeader
   173→        title="Order Cart"
   174→        description="Submitting sends a request — Jeffery Group confirms availability, pricing and the delivery date before anything is produced or shipped."
   175→      />
   176→
   177→      {/* Stepper */}
   178→      <ol className="mb-5 flex flex-wrap items-center gap-x-1 gap-y-2" aria-label="Order request steps">
   179→        {STEPS.map((s, i) => {
   180→          const done = s.n < step;
   181→          const current = s.n === step;
   182→          return (
   183→            <li key={s.n} className="flex items-center gap-1.5 sm:gap-2">
   184→              <button
   185→                type="button"
   186→                disabled={s.n >= step}
   187→                onClick={() => goTo(s.n)}
   188→                aria-current={current ? "step" : undefined}
   189→                className={cn(
   190→                  "flex items-center gap-2 rounded-full py-1 pl-1 pr-2.5 text-[12.5px] transition-colors",
   191→                  s.n >= step ? "cursor-default" : "hover:bg-muted/60",
   192→                  current ? "font-medium text-foreground" : done ? "text-muted-foreground" : "text-muted-foreground/70",
   193→                )}
   194→              >
   195→                <span
   196→                  className={cn(
   197→                    "tnum flex size-6 items-center justify-center rounded-full border text-[11px] font-semibold",
   198→                    done && "border-primary bg-primary text-primary-foreground",
   199→                    current && "border-primary bg-primary/[0.08] text-primary",
   200→                    !done && !current && "border-border bg-background text-muted-foreground",
   201→                  )}
   202→                >
   203→                  {done ? <Check className="size-3.5" aria-hidden /> : s.n}
   204→                </span>
   205→                {s.label}
   206→              </button>
   207→              {i < STEPS.length - 1 && <span className="h-px w-4 bg-border sm:w-7" aria-hidden />}
   208→            </li>
   209→          );
   210→        })}
   211→      </ol>
   212→
   213→      <form onSubmit={submit} className="space-y-5">
   214→        {/* ── Step 1: Review lines ── */}
   215→        {step === 1 && (
   216→          <SectionCard
   217→            title="Review items"
   218→            description="Quantities respect each product's minimum and multiples."
   219→            contentClassName="p-0 sm:p-0"
   220→          >
   221→            {/* Desktop table */}
   222→            <div className="hidden md:block">
   223→              <Table>
   224→                <TableHeader>
   225→                  <TableRow className="hover:bg-transparent">
   226→                    <TableHead className="w-[38%]">Product</TableHead>
   227→                    <TableHead className="text-right">Unit price</TableHead>
   228→                    <TableHead className="text-center">Quantity</TableHead>
   229→                    <TableHead className="text-right">Line total</TableHead>
   230→                    <TableHead className="w-12" />
   231→                  </TableRow>
   232→                </TableHeader>
   233→                <TableBody>
   234→                  {lines.map((l) => (
   235→                    <CartRow key={l.productId} line={l} setQty={setQty} removeLine={removeLine} />
   236→                  ))}
   237→                </TableBody>
   238→              </Table>
   239→            </div>
   240→            {/* Mobile stacked cards */}
   241→            <ul className="divide-y md:hidden">
   242→              {lines.map((l) => (
   243→                <li key={l.productId} className="flex flex-col gap-3 px-4 py-4">
   244→                  <div className="flex items-start justify-between gap-3">
   245→                    <div className="flex min-w-0 items-center gap-3">
   246→                      <ProductArtwork sku={l.sku} compact />
   247→                      <div className="min-w-0">
   248→                        <p className="truncate text-[13.5px] font-medium leading-snug">{l.name}</p>
   249→                        <Mono className="text-muted-foreground">{l.sku}</Mono>
   250→                      </div>
   251→                    </div>
   252→                    <Button
   253→                      variant="ghost"
   254→                      size="icon"
   255→                      className="size-8 shrink-0 text-muted-foreground hover:text-red-600"
   256→                      onClick={() => removeLine(l.productId)}
   257→                      aria-label={`Remove ${l.name} from cart`}
   258→                    >
   259→                      <Trash2 className="size-4" aria-hidden />
   260→                    </Button>
   261→                  </div>
   262→                  <div className="flex items-center justify-between gap-3">
   263→                    <QtyStepper
   264→                      value={l.qty}
   265→                      onChange={(q) => setQty(l.productId, q)}
   266→                      min={l.minOrderQty}
   267→                      increment={l.increment}
   268→                      size="sm"
   269→                      ariaLabel={`Quantity of ${l.name}`}
   270→                    />
   271→                    <div className="text-right">
   272→                      <Money cents={l.qty * l.unitPriceCents} className="text-[14px] font-semibold" />
   273→                      <p className="text-[11.5px] text-muted-foreground">
   274→                        <Money cents={l.unitPriceCents} muted /> per {l.unit}
   275→                      </p>
   276→                    </div>
   277→                  </div>
   278→                </li>
   279→              ))}
   280→            </ul>
   281→
   282→            <div className="border-t bg-muted/20 px-4 py-4 sm:px-5">
   283→              <CartTotals subtotal={subtotal} tax={tax} total={total} />
   284→            </div>
   285→          </SectionCard>
   286→        )}
   287→
   288→        {/* ── Step 2: Delivery ── */}
   289→        {step === 2 && (
   290→          <SectionCard title="Delivery details" description="Where should this order ship, and anything we should know?">
   291→            {accountError ? (
   292→              <ErrorAlert message="We could not load your shipping addresses." />
   293→            ) : accountLoading || !account ? (
   294→              <div className="space-y-3">
   295→                <Skeleton className="h-10 w-full" />
   296→                <Skeleton className="h-16 w-full" />
   297→                <Skeleton className="h-20 w-full" />
   298→              </div>
   299→            ) : (
   300→              <div className="space-y-4">
   301→                {addresses.length === 0 ? (
   302→                  <div className="rounded-md border border-amber-200 bg-amber-50/70 px-4 py-3 text-[13px] text-amber-800">
   303→                    No shipping addresses on file yet — add one in Settings before submitting.
   304→                  </div>
   305→                ) : (
   306→                  <>
   307→                    <div className="space-y-2">
   308→                      <Label htmlFor="cart-address">Shipping address</Label>
   309→                      <Select
   310→                        value={selectedAddress?.id ?? ""}
   311→                        onValueChange={(v) => setAddressId(v)}
   312→                      >
   313→                        <SelectTrigger id="cart-address" className="w-full sm:w-[380px]">
   314→                          <SelectValue placeholder="Choose an address" />
   315→                        </SelectTrigger>
   316→                        <SelectContent>
   317→                          {addresses.map((a) => (
   318→                            <SelectItem key={a.id} value={a.id}>
   319→                              {a.label}
   320→                              {a.isDefault ? " (default)" : ""}
   321→                            </SelectItem>
   322→                          ))}
   323→                        </SelectContent>
   324→                      </Select>
   325→                    </div>
   326→                    {selectedAddress && (
   327→                      <div className="rounded-md border bg-muted/30 px-4 py-3">
   328→                        <p className="flex items-center gap-1.5 text-[12px] font-medium uppercase tracking-[0.06em] text-muted-foreground">
   329→                          <Truck className="size-3.5" aria-hidden /> Ships to
   330→                        </p>
   331→                        <p className="mt-1.5 text-[13.5px] font-medium">{selectedAddress.label}</p>
   332→                        <p className="mt-0.5 text-[13px] text-muted-foreground">
   333→                          {selectedAddress.line1}
   334→                          {selectedAddress.line2 ? `, ${selectedAddress.line2}` : ""}
   335→                          <br />
   336→                          {selectedAddress.city}, {selectedAddress.province} {selectedAddress.postal}
   337→                        </p>
   338→                      </div>
   339→                    )}
   340→                    <button
   341→                      type="button"
   342→                      className="text-[12.5px] font-medium text-muted-foreground underline-offset-2 transition-colors hover:text-foreground hover:underline"
   343→                      onClick={() => navigate("settings")}
   344→                    >
   345→                      Manage addresses in Settings
   346→                    </button>
   347→                  </>
   348→                )}
   349→                <div className="space-y-2">
   350→                  <Label htmlFor="cart-notes">Order notes</Label>
   351→                  <Textarea
   352→                    id="cart-notes"
   353→                    rows={3}
   354→                    maxLength={2000}
   355→                    placeholder="Anything our order desk should know — dock hours, receiving contacts, packaging preferences…"
   356→                    value={orderNotes}
   357→                    onChange={(e) => setOrderNotes(e.target.value)}
   358→                  />
   359→                </div>
   360→                <div className="max-w-md space-y-2">
   361→                  <Label htmlFor="cart-delivery">Requested delivery</Label>
   362→                  <Input
   363→                    id="cart-delivery"
   364→                    maxLength={200}
   365→                    placeholder="e.g. Before month end, or a specific date"
   366→                    value={requestedDelivery}
   367→                    onChange={(e) => setRequestedDelivery(e.target.value)}
   368→                  />
   369→                  <p className="text-[12px] text-muted-foreground">
   370→                    Requested only — we&apos;ll confirm the actual date when the order is confirmed.
   371→                  </p>
   372→                </div>
   373→              </div>
   374→            )}
   375→          </SectionCard>
   376→        )}
   377→
   378→        {/* ── Step 3: Reference ── */}
   379→        {step === 3 && (
   380→          <SectionCard title="Reference details" description="Optional — helps you match this request to your own records.">
   381→            <div className="space-y-4">
   382→              <div className="max-w-md space-y-2">
   383→                <Label htmlFor="cart-po">PO reference</Label>
   384→                <Input
   385→                  id="cart-po"
   386→                  maxLength={60}
   387→                  placeholder="e.g. MLP-PO-0918"
   388→                  value={poNumber}
   389→                  onChange={(e) => setPoNumber(e.target.value)}
   390→                />
   391→                <p className="text-[12px] text-muted-foreground">Your purchase order number, if one applies.</p>
   392→              </div>
   393→              <div className="space-y-2">
   394→                <Label htmlFor="cart-special">Special instructions</Label>
   395→                <Textarea
   396→                  id="cart-special"
   397→                  rows={4}
   398→                  maxLength={2000}
   399→                  placeholder="Handling, packing or delivery preferences beyond the order notes…"
   400→                  value={specialInstructions}
   401→                  onChange={(e) => setSpecialInstructions(e.target.value)}
   402→                />
   403→              </div>
   404→            </div>
   405→          </SectionCard>
   406→        )}
   407→
   408→        {/* ── Step 4: Review & submit ── */}
   409→        {step === 4 && (
   410→          <div className="grid grid-cols-1 gap-5 lg:grid-cols-[minmax(0,1fr)_360px]">
   411→            <SectionCard title="Final review" description="Check everything before submitting your request." contentClassName="p-0 sm:p-0">
   412→              <ul className="divide-y">
   413→                {lines.map((l) => (
   414→                  <li key={l.productId} className="flex items-center gap-3 px-4 py-3 sm:px-5">
   415→                    <ProductArtwork sku={l.sku} compact className="hidden sm:block" />
   416→                    <div className="min-w-0 flex-1">
   417→                      <p className="truncate text-[13.5px] font-medium">{l.name}</p>
   418→                      <p className="text-[12px] text-muted-foreground">
   419→                        <Mono>{l.sku}</Mono> · <span className="tnum">{l.qty.toLocaleString()}</span> ×{" "}
   420→                        <Money cents={l.unitPriceCents} muted />
   421→                      </p>
   422→                    </div>
   423→                    <Money cents={l.qty * l.unitPriceCents} className="text-[13.5px] font-medium" />
   424→                  </li>
   425→                ))}
   426→              </ul>
   427→              <div className="border-t bg-muted/20 px-4 py-4 sm:px-5">
   428→                <CartTotals subtotal={subtotal} tax={tax} total={total} />
   429→              </div>
   430→            </SectionCard>
   431→
   432→            <SectionCard title="Request summary">
   433→              <dl className="m-0 space-y-3 text-[13px]">
   434→                <div>
   435→                  <dt className="text-[12px] font-medium uppercase tracking-[0.06em] text-muted-foreground">Items</dt>
   436→                  <dd className="mt-0.5 text-foreground">
   437→                    {lines.length} product{lines.length === 1 ? "" : "s"} ·{" "}
   438→                    {lines.reduce((s, l) => s + l.qty, 0).toLocaleString()} units
   439→                  </dd>
   440→                </div>
   441→                <div>
   442→                  <dt className="text-[12px] font-medium uppercase tracking-[0.06em] text-muted-foreground">Ship to</dt>
   443→                  <dd className="mt-0.5 text-foreground">
   444→                    {selectedAddress ? (
   445→                      <>
   446→                        {selectedAddress.label}
   447→                        <span className="block text-muted-foreground">
   448→                          {selectedAddress.line1}
   449→                          {selectedAddress.line2 ? `, ${selectedAddress.line2}` : ""}
   450→                          {", "}
   451→                          {selectedAddress.city}, {selectedAddress.province} {selectedAddress.postal}
   452→                        </span>
   453→                      </>
   454→                    ) : (
   455→                      <span className="text-muted-foreground">No address selected</span>
   456→                    )}
   457→                  </dd>
   458→                </div>
   459→                {requestedDelivery.trim() && (
   460→                  <div>
   461→                    <dt className="text-[12px] font-medium uppercase tracking-[0.06em] text-muted-foreground">Requested delivery</dt>
   462→                    <dd className="mt-0.5 text-foreground">{requestedDelivery.trim()}</dd>
   463→                  </div>
   464→                )}
   465→                {poNumber.trim() && (
   466→                  <div>
   467→                    <dt className="text-[12px] font-medium uppercase tracking-[0.06em] text-muted-foreground">PO reference</dt>
   468→                    <dd className="mt-0.5 text-foreground">{poNumber.trim()}</dd>
   469→                  </div>
   470→                )}
   471→                {(orderNotes.trim() || specialInstructions.trim()) && (
   472→                  <div>
   473→                    <dt className="text-[12px] font-medium uppercase tracking-[0.06em] text-muted-foreground">Notes</dt>
   474→                    <dd className="mt-0.5 whitespace-pre-line text-muted-foreground">
   475→                      {[orderNotes.trim() && `Order notes: ${orderNotes.trim()}`, specialInstructions.trim() && `Special instructions: ${specialInstructions.trim()}`]
   476→                        .filter(Boolean)
   477→                        .join("\n\n")}
   478→                    </dd>
   479→                  </div>
   480→                )}
   481→              </dl>
   482→
   483→              <Separator className="my-4" />
   484→
   485→              {submitError && (
   486→                <div role="alert" className="mb-3 rounded-md border border-red-200 bg-red-50 px-3.5 py-2.5 text-[13px] text-red-700">
   487→                  {submitError}
   488→                </div>
   489→              )}
   490→              <Button
   491→                type="submit"
   492→                className="h-10 w-full gap-2"
   493→                disabled={submitMutation.isPending || addresses.length === 0}
   494→              >
   495→                {submitMutation.isPending ? (
   496→                  <>
   497→                    <Loader2 className="size-4 animate-spin" aria-hidden /> Submitting…
   498→                  </>
   499→                ) : (
   500→                  <>
   501→                    <Check className="size-4" aria-hidden /> Submit order request
   502→                  </>
   503→                )}
   504→              </Button>
   505→              <p className="mt-3 text-[12px] leading-relaxed text-muted-foreground">
   506→                No goods are promised or reserved at this point. Our order desk reviews availability and
   507→                pricing, then confirms with you before production.
   508→              </p>
   509→            </SectionCard>
   510→          </div>
   511→        )}
   512→
   513→        {/* Step navigation */}
   514→        <div className="flex items-center justify-between gap-3">
   515→          {step > 1 ? (
   516→            <Button
   517→              type="button"
   518→              variant="outline"
   519→              className="gap-1.5"
   520→              onClick={() => goTo(step - 1)}
   521→              disabled={submitMutation.isPending}
   522→            >
   523→              <ArrowLeft className="size-4" aria-hidden /> Back
   524→            </Button>
   525→          ) : (
   526→            <span />
   527→          )}
