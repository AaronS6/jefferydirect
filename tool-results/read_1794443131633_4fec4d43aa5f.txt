   463→function ProductDialog({
   464→  categories,
   465→  product,
   466→  open,
   467→  onOpenChange,
   468→}: {
   469→  categories: CategoryDto[];
   470→  product?: AdminProduct;
   471→  open: boolean;
   472→  onOpenChange: (v: boolean) => void;
   473→}) {
   474→  const queryClient = useQueryClient();
   475→  const isEdit = Boolean(product);
   476→
   477→  const [form, setForm] = useState<ProductForm>(() =>
   478→    product
   479→      ? {
   480→          sku: product.sku,
   481→          name: product.name,
   482→          categoryId: product.categoryId,
   483→          shortDescription: product.shortDescription,
   484→          description: product.description,
   485→          material: product.material ?? "",
   486→          dimensions: product.dimensions ?? "",
   487→          colour: product.colour ?? "",
   488→          printingOptions: product.printingOptions ?? "",
   489→          casePack: String(product.casePack),
   490→          palletPack: product.palletPack != null ? String(product.palletPack) : "",
   491→          minOrderQty: String(product.minOrderQty),
   492→          increment: String(product.increment),
   493→          unit: product.unit,
   494→          defaultPrice: product.defaultPriceCents != null ? centsToInputValue(product.defaultPriceCents) : "",
   495→          stockStatus: product.stockStatus,
   496→          leadTimeNote: product.leadTimeNote ?? "",
   497→        }
   498→      : { ...EMPTY_PRODUCT_FORM },
   499→  );
   500→  const [submitError, setSubmitError] = useState<string | null>(null);
   501→
   502→  const parsedPrice = form.defaultPrice.trim() === "" ? null : parseDollarsToCents(form.defaultPrice);
   503→  const priceInvalid = form.defaultPrice.trim() !== "" && parsedPrice === null;
   504→
   505→  const save = useMutation<{ ok: boolean } | { id: string }, Error>({
   506→    mutationFn: async (): Promise<{ ok: boolean } | { id: string }> => {
   507→      const payload = {
   508→        sku: form.sku.trim(),
   509→        name: form.name.trim(),
   510→        categoryId: form.categoryId,
   511→        shortDescription: form.shortDescription.trim(),
   512→        description: form.description.trim(),
   513→        material: form.material.trim() || null,
   514→        dimensions: form.dimensions.trim() || null,
   515→        colour: form.colour.trim() || null,
   516→        printingOptions: form.printingOptions.trim() || null,
   517→        casePack: Number.parseInt(form.casePack, 10),
   518→        palletPack: form.palletPack.trim() === "" ? null : Number.parseInt(form.palletPack, 10),
   519→        minOrderQty: Number.parseInt(form.minOrderQty, 10),
   520→        increment: Number.parseInt(form.increment, 10),
   521→        unit: form.unit,
   522→        defaultPriceCents: form.defaultPrice.trim() === "" ? null : parsedPrice,
   523→        stockStatus: form.stockStatus,
   524→        leadTimeNote: form.leadTimeNote.trim() || null,
   525→      };
   526→      return isEdit
   527→        ? api.put<{ ok: boolean }>(`/api/admin/products/${product!.id}`, payload)
   528→        : api.post<{ id: string }>("/api/admin/products", payload);
   529→    },
   530→    onSuccess: () => {
   531→      queryClient.invalidateQueries({ queryKey: ["admin", "products"] });
   532→      queryClient.invalidateQueries({ queryKey: ["admin", "pricing"] });
   533→      queryClient.invalidateQueries({ queryKey: ["admin", "overview"] });
   534→      onOpenChange(false);
   535→      toast({
   536→        title: isEdit ? "Product updated" : "Product created",
   537→      });
   538→    },
   539→    onError: (err: Error) => setSubmitError(err.message),
   540→  });
   541→
   542→  const set =
   543→    (key: keyof ProductForm) =>
   544→    (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) =>
   545→      setForm((f) => ({ ...f, [key]: e.target.value }));
   546→
   547→  function submit(e: React.FormEvent) {
   548→    e.preventDefault();
   549→    setSubmitError(null);
   550→    if (!form.sku.trim() || !form.name.trim() || !form.categoryId || !form.shortDescription.trim() || !form.description.trim()) {
   551→      setSubmitError("SKU, name, category and both descriptions are required.");
   552→      return;
   553→    }
   554→    for (const [label, raw] of [
   555→      ["Case pack", form.casePack],
   556→      ["Minimum order quantity", form.minOrderQty],
   557→      ["Quantity increment", form.increment],
   558→    ] as const) {
   559→      const n = Number.parseInt(raw, 10);
   560→      if (!Number.isFinite(n) || n < 1) {
   561→        setSubmitError(`${label} must be a whole number of 1 or more.`);
   562→        return;
