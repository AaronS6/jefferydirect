    50→// ─── Builder draft state ─────────────────────────────────────────────────────
    51→
    52→type DraftRow = { uid: string; description: string; kind: string; qty: string; price: string };
    53→
    54→let uidCounter = 0;
    55→const newUid = () => `row-${++uidCounter}`;
    56→
    57→function rowsFromQuote(quote: QuotationDetail): DraftRow[] {
    58→  if (quote.items.length === 0) return [emptyRow()];
    59→  return quote.items.map((i) => ({
    60→    uid: newUid(),
    61→    description: i.description,
    62→    kind: i.kind,
    63→    qty: String(i.qty),
    64→    price: centsToInputValue(i.unitPriceCents),
    65→  }));
    66→}
    67→
    68→function emptyRow(): DraftRow {
    69→  return { uid: newUid(), description: "", kind: "PRODUCT", qty: "", price: "" };
    70→}
    71→
    72→function rowCents(row: DraftRow): number | null {
    73→  const cents = parseDollarsToCents(row.price.trim());
    74→  const qty = Number.parseInt(row.qty, 10);
    75→  if (cents === null || !Number.isFinite(qty) || qty <= 0) return null;
    76→  return cents * qty;
    77→}
    78→
    79→function rowIssues(row: DraftRow): { description: boolean; qty: boolean; price: boolean } {
    80→  const cents = parseDollarsToCents(row.price.trim());
    81→  const qty = Number.parseInt(row.qty, 10);
    82→  return {
    83→    description: row.description.trim() === "",
    84→    qty: !Number.isFinite(qty) || qty <= 0,
    85→    price: cents === null,
    86→  };
    87→}

   149→        items,
   150→        expiryDate: expiry || null,
   151→        terms: terms.trim() || null,
   152→        internalNote: internalNote.trim() || null,
   153→      });

   166→      <div>

   170→    );

   172→

   175→      <div>
   176→        <PageHeader title="Quote" onBack={back} />

   178→          <SectionCard>
   179→            <RowsSkeleton rows={5} />
   180→          </SectionCard>
   181→          <SectionCard>
   182→            <RowsSkeleton rows={4} />
   183→          </SectionCard>
   184→        </div>
   185→      </div>
   186→    );
   187→  }

   190→  const reissuable = quote.status === "ISSUED" || quote.status === "REVISED" || quote.status === "EXPIRED";
   191→  const responded = quote.status === "ACCEPTED" || quote.status === "REJECTED";
   192→  const hasItems = quote.items.length > 0;
   193→

   202→  const rowsValid = rows.length > 0 && rows.every((r) => {

   225→              <Button
   226→                size="sm"
   227→                variant="outline"

   229→                disabled={quote.status !== "REQUESTED" || acknowledge.isPending}
   230→                onClick={() => acknowledge.mutate()}
   231→              >
   232→                {acknowledge.isPending ? <Loader2 className="size-3.5 animate-spin" aria-hidden /> : <CheckCircle2 className="size-3.5" aria-hidden />}
   233→                Acknowledge

   235→            )}

   237→          </div>
   238→        }

   240→
   241→      <ViewEnter>
   242→      <div className="grid items-start gap-4 sm:gap-6 lg:grid-cols-[minmax(0,1fr)_360px]">

   244→        <div className="min-w-0 space-y-4 sm:space-y-6">
   245→          <SectionCard title="Customer's request">
   246→            <dl>
   247→              <DefRow label="Category" value={dashed(quote.category)} />
   248→              <DefRow label="Product" value={dashed(quote.productName)} />
   249→              <DefRow label="Dimensions" value={dashed(quote.dimensions)} />

   252→              <DefRow label="Printing" value={dashed(quote.printing)} />

   254→              <DefRow label="Needed by" value={dashed(quote.neededBy)} />
   255→              <DefRow label="Destination" value={dashed(quote.destination)} />
   256→              <DefRow label="Specifications" value={dashed(quote.specs)} />
   257→              <DefRow label="Comments" value={dashed(quote.comments)} />
   258→            </dl>

   260→
   261→          {/* Response zone */}
   262→          {responded && (
   263→            <div

   282→                >
   283→                  Customer {quote.status === "ACCEPTED" ? "accepted" : "declined"} this quote

   286→                {quote.respondedNote && (
   287→                  <p className="mt-1 text-body-sm leading-relaxed text-foreground/80">{quote.respondedNote}</p>

   290→            </div>

   298→                response. Re-issue to send fresh pricing.

   307→              action={

   309→                  <Button variant="outline" size="sm" className="gap-1.5" onClick={() => setPreviewOpen(true)}>
   310→                    <Eye className="size-3.5" aria-hidden /> Preview document

   312→                  {reissuable && (

   314→                      variant={preIssue ? "default" : "outline"}

   316→                      className="gap-1.5"
   317→                      onClick={() => {
   318→                        setBuilderOpen(false);

   321→                    >
   322→                      <RefreshCcw className="size-3.5" aria-hidden /> Re-issue (revision)

   326→              }
   327→            >
   328→              <Table>
   329→                <TableHeader>
   330→                  <TableRow>
   331→                    <Th>Description</Th>
   332→                    <Th>Type</Th>
   333→                    <Th className="text-right">Qty</Th>
   334→                    <Th className="text-right">Unit</Th>
   335→                    <Th className="text-right">Amount</Th>
   336→                  </TableRow>
   337→                </TableHeader>
   338→                <TableBody>
   339→                  {quote.items.map((item) => (
   340→                    <TableRow key={item.id}>
   341→                      <Td className="max-w-[320px] whitespace-normal">{item.description}</Td>
   342→                      <Td>

   344→                          {QUOTE_ITEM_KIND_LABELS[item.kind] ?? item.kind}
   345→                        </span>
   346→                      </Td>
   347→                      <Td className="tnum text-right">{item.qty.toLocaleString()}</Td>
   348→                      <Td className="text-right">
   349→                        <Money cents={item.unitPriceCents} />
   350→                      </Td>
   351→                      <Td className="text-right font-medium">
   352→                        <Money cents={item.lineTotalCents} />
   353→                      </Td>
   354→                    </TableRow>
   355→                  ))}
   356→                </TableBody>
   357→                <TableFooter>
   358→                  <TableRow>
   359→                    <Td className="text-right font-medium">Quoted total</Td>
   360→                    <Td />
   361→                    <Td />
   362→                    <Td />
   363→                    <Td className="text-right font-semibold">
   364→                      <MoneyWithCode cents={quote.itemsTotalCents} />
   365→                    </Td>
