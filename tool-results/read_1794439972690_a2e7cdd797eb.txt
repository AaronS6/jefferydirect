    81→    81→// Ink dots for completed steps, kraft ring on the current one, hollow future.
    82→    82→function StatusStepper({ status, history }: { status: string; history: OrderHistoryEntry[] }) {
    83→    83→  const flow = ORDER_STATUSES;
    84→    84→  const terminal = ORDER_TERMINAL_STATUSES.includes(status);
    85→    85→  const currentIdx = flow.indexOf(status as (typeof flow)[number]);
    86→    86→
    87→    87→  // Latest history entry per status (ascending order).
    88→    88→  const latestByStatus = new Map<string, OrderHistoryEntry>();
    89→    89→  for (const h of [...history].sort((a, b) => a.createdAt.localeCompare(b.createdAt))) {
    90→    90→    latestByStatus.set(h.status, h);
    91→    91→  }
    92→    92→
    93→    93→  const progress =
    94→    94→    currentIdx >= 0 ? Math.round(((currentIdx + 1) / flow.length) * 100) : 100;
    95→    95→  const dead = status === "CANCELLED" || status === "REJECTED";
    96→    96→
    97→    97→  return (
    98→    98→    <>
    99→    99→      {/* Compact progress — below xl */}
   100→   100→      <div className="flex flex-wrap items-center gap-x-4 gap-y-2.5 xl:hidden" aria-label="Order progress">
   101→   101→        <span
   102→   102→          aria-hidden
   103→   103→          className={cn(
   104→   104→            "flex size-[14px] shrink-0 items-center justify-center rounded-full",
   105→   105→            dead
   106→   106→              ? "border-2 border-[var(--status-danger-dot)] bg-[var(--status-danger-bg)]"
   107→   107→              : terminal
   108→   108→                ? "bg-foreground"
   109→   109→                : "border-2 border-kraft-strong bg-kraft",
   110→   110→          )}
   111→   111→        >
   112→   112→          {!dead && !terminal && <span className="size-1.5 rounded-full bg-background" aria-hidden />}
   113→   113→          {terminal && <span className="size-[4.5px] rounded-full bg-background" aria-hidden />}
   114→   114→        </span>
   115→   115→        <p className="text-body font-semibold text-foreground">
   116→   116→          {ORDER_STATUS_LABELS[status] ?? status}
   117→   117→        </p>
   118→   118→        {!dead && (
   119→   119→          <p className="text-caption text-muted-foreground">
   120→   120→            <span className="tnum">Step {Math.max(1, currentIdx + 1)} of {flow.length}</span>
   121→   121→          </p>
   122→   122→        )}
   123→   123→        <div className="h-1.5 min-w-24 flex-1 overflow-hidden rounded-full bg-secondary" aria-hidden>
   124→   124→          <div
   125→   125→            className={cn("h-full rounded-full transition-[width] duration-500", dead ? "bg-[var(--status-danger-dot)]" : "bg-kraft-strong")}
   126→   126→            style={{ width: `${dead ? 100 : progress}%` }}
   127→   127→          />
   128→   128→        </div>
   129→   129→      </div>
   130→   130→
   131→   131→      {/* Full rail — xl and up */}
   132→   132→      <ol className="hidden items-start xl:flex" aria-label="Order progress steps">
   133→   133→        {flow.map((s, i) => {
   134→   134→          const entry = latestByStatus.get(s);
   135→   135→          const done = i < currentIdx || (terminal && entry !== undefined);
   136→   136→          const current = i === currentIdx && !terminal;
   137→   137→          return (
   138→   138→            <li
   139→   139→              key={s}
   140→   140→              className="relative flex min-w-[96px] flex-1 basis-0 flex-col items-center gap-2.5 px-1 text-center"
   141→   141→            >
   142→   142→              {i < flow.length - 1 && (
   143→   143→                <span
   144→   144→                  aria-hidden
   145→   145→                  className={cn(
   146→   146→                    "absolute top-[7px] left-[calc(50%+10px)] h-px w-[calc(100%-20px)]",
   147→   147→                    done || current ? "bg-foreground/25" : "bg-border",
   148→   148→                  )}
   149→   149→                />
   150→   150→              )}
   151→   151→              <span
   152→   152→                className={cn(
   153→   153→                  "relative z-10 flex size-[14px] shrink-0 items-center justify-center rounded-full",
   154→   154→                  done
   155→   155→                    ? "bg-foreground"
   156→   156→                    : current
   157→   157→                      ? "border-2 border-kraft-strong bg-kraft"
   158→   158→                      : "border border-border bg-card",
   159→   159→                )}
   160→   160→              >
   161→   161→                {done && <span className="size-[4.5px] rounded-full bg-background" aria-hidden />}
   162→   162→              </span>
   163→   163→              <div className="min-w-0 max-w-full">
   164→   164→                <p
   165→   165→                  className={cn(

   167→   167→                    current

   169→   169→                      : done
   170→   170→                        ? "font-medium text-foreground/80"
   171→   171→                        : "font-medium text-muted-foreground",
   172→   172→                  )}
   173→   173→                >
   174→   174→                  {ORDER_STATUS_LABELS[s]}
   175→   175→                </p>
